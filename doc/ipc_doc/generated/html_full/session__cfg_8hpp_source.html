30/**
31 * Grouping of compile-time config/knobs (as of this writing: constants) governing some of the behavior of
32 * ipc::session sessions.
33 *
34 * ### Rationale ###
35 * Some of these could be viewed as implementation details; nevertheless they are public, much like
36 * transport::Native_socket_stream_cfg: it is harmless, and it can be helpful for logging/reporting or for
37 * understanding the system's behavior (e.g., how quickly a stuck opposing process is detected).
38 *
39 * @todo Session_cfg is the nucleus of a more general overhaul of ipc::session configuration (e.g., grouping
40 * various `Session` / `Session_server` template parameters into a `struct`; possibly per-Client_app knobs).
41 * Its constants may then become run-time and/or per-session-configurable.
42 */
43class Session_cfg
44{
45public:
46 // Constants.
47
48 /**
49 * The period with which each session-internal channel -- notably the session master channel; also the
50 * internal channel used by an arena-lending SHM-provider (as of this writing SHM-jemalloc) -- sends low-level
51 * keep-alive pings to the opposing peer. See transport::Channel::auto_ping().
52 *
53 * The opposing side uses the same value; hence #S_IDLE_TIMEOUT (which must comfortably exceed it) works
54 * consistently in both directions.
55 */
56 static constexpr util::Fine_duration S_PING_PERIOD = boost::chrono::seconds{2};
57
58 /**
59 * The idle timeout of each session-internal channel (see #S_PING_PERIOD): if no low-level traffic -- including
60 * the opposing side's pings -- arrives in this long, the channel is considered hosed; and hence the session is
61 * too. See transport::Channel::idle_timer_run().
62 *
63 * Its purpose is to detect an opposing process that is stuck (alive but not functioning), as opposed to dead
64 * (detected immediately regardless). As a false positive (hosing a healthy session over a transient stall)
65 * is costlier than slower detection of a stuck peer, the value is generous relative to #S_PING_PERIOD.
66 */
67 static constexpr util::Fine_duration S_IDLE_TIMEOUT = boost::chrono::seconds{15};
68
69private:
70 // Constructors/destructor.
71
72 /// Forbid instantiation, for it is pointless.
73 Session_cfg() = delete;
74}; // class Session_cfg

76} // namespace ipc::session

30/* (File-local/internal-use) Constructs a payload-size histogram whose last bucket isolates exactly max_payload_size.
31 * Bucket 0 absorbs the remainder when max_payload_size is not divisible by bucket_sz,
32 * so that subsequent bucket boundaries align and one falls exactly on max_payload_size.
33 * When divisible, bucket 0 is the same width as the rest (uniform). */
34flow::util::stat::Histogram_counter make_payload_sz_histo(size_t max_payload_size, size_t bucket_sz)
35{
36 assert((max_payload_size >= 1) && "max_payload_size must be >= 1.");
37
38 const auto remainder = max_payload_size % bucket_sz;
39 const auto bucket0_sz = (remainder == 0) ? bucket_sz : remainder;
40
41 return flow::util::stat::Histogram_counter{size_t(2) + ((max_payload_size - size_t(1)) / bucket_sz),
42 int64_t(bucket0_sz),
43 int64_t(bucket_sz), 0};
44}
45
46} // namespace (anon)

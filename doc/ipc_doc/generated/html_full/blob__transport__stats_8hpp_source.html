32/**
33 * Cumulative stats from the outgoing-direction (send) pipe of a Blob_sender or Native_handle_sender
34 * concept implementation. Applicable to any such implementation; also counts native-handle sends if applicable.
35 *
36 * @note While internally a payload-to-be-sent may have to be queued on would-block from the low-level transport
37 * (e.g., kernel send buffer full in `Native_socket_stream`), counters like #m_total_msgs and #m_total_bytes
38 * and #m_msgs_with_hndls count the full payload ASAP without waiting for all of it to be "really sent."
39 * Meanwhile #m_would_block_count and #m_snd_q_hi_wmark can give a sense of how often the internal queuing
40 * had to be invoked.
41 */
42struct Blob_snd_stats
43{
44 // Constants.
45
46 /// Payload-size histogram bucket width (bytes). The last bucket isolates the max payload size.
47 static constexpr size_t S_PAYLOAD_SZ_HISTO_BUCKET_SZ = 1024;
48
49 // Data.
50
51 /// Total user messages successfully passed to the low-level send path via `send_*()`.
52 uint64_t m_total_msgs = 0;
53
54 /// Total user blob bytes sent (sum of blob sizes from `send_*()`).
55 uint64_t m_total_bytes = 0;
56
57 /// Total low-level bytes sent (user blob bytes plus protocol headers and framing).
58 uint64_t m_total_low_lvl_bytes = 0;
59
60 /// Number of messages sent that carried at least one native handle. Currently at most 1 handle per message.
61 uint64_t m_msgs_with_hndls = 0;
62
63 /// Number of times a would-block condition required payload queueing (low-level transport's internal buffer full).
64 uint64_t m_would_block_count = 0;
65
66 /// Number of auto-pings sent (initial from `auto_ping()` plus periodic timer-fired pings).
67 uint64_t m_auto_pings = 0;
68
69 /// Current depth of the pending-payloads send queue (queue added-to when would-block occurs).
70 size_t m_snd_q_depth = 0;
71
72 /// Max of #m_snd_q_depth observed so far.
73 size_t m_snd_q_hi_wmark = 0;
74
75 /**
76 * Payload-size histogram for sent user messages. Maxed-out size gets its own bucket.
77 *
78 * @todo Consider replacing #m_histo_payload_sz's uniform low end with log-scale buckets (64, 128, ...,
79 * 1Ki), if field experience keeps wanting better resolution among small messages: bucket 0 currently
80 * swallows all sub-1Ki traffic. (The latter includes ~all outer messages of SHM-backed *structured* channels;
81 * conceivably it would be nice to capture those, for applications that use that feature; which to be clear
82 * means (1) using struc::Channel or struc::sync_io::Channel in ther first place; and given that also (2) using
83 * the SHM-backed variety as opposed to merely heap-backed.)
84 */
85 flow::util::stat::Histogram_counter m_histo_payload_sz;
86
87 // Constructors.
88
89 /**
90 * Constructs stats with all counters zeroed and a payload-size histogram sized for the given max payload.
91 * The last bucket isolates exactly `max_payload_size`. If `max_payload_size` is not divisible by
92 * #S_PAYLOAD_SZ_HISTO_BUCKET_SZ, bucket 0 absorbs the remainder so that subsequent bucket boundaries
93 * align and one falls exactly on `max_payload_size`.
94 *
95 * @param max_payload_size
96 * Maximum blob payload size for this transport (e.g., `Native_socket_stream_cfg::S_MAX_META_BLOB_LENGTH`).
97 * Must be >= 1.
98 */
99 explicit Blob_snd_stats(size_t max_payload_size);
100}; // struct Blob_snd_stats

172 *

174 * Maximum blob payload size for this transport (e.g., `Native_socket_stream_cfg::S_MAX_META_BLOB_LENGTH`).
175 * Must be >= 1.
176 */

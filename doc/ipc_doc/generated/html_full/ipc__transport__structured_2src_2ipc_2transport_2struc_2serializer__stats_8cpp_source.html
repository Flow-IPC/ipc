38flow::util::stat::Histogram_counter Histo_cfg::build() const
39{
40 using flow::util::stat::Histogram_counter;
41 using value_t = Histogram_counter::value_t;
42 using std::vector;
43
44 if (m_ratio == 0)
45 {
46 // Linear scale.
47 return Histogram_counter{m_n_bkts, value_t(m_bkt0), value_t(m_bkt), 0};
48 }
49 // else: Geometric scale. Expand the ladder into the arbitrary-bucket-widths ctor form.
50
51 assert((m_ratio >= 2) && (m_n_bkts >= 2) && (m_bkt0 >= 1) && (m_bkt == 0)
52 && "Geometric-scale Histo_cfg: require ratio >= 2, n_bkts >= 2, floor >= 1, m_bkt zero (unused).");
53
54 vector<value_t> bkt_val0s(m_n_bkts);
55 bkt_val0s[0] = 0; // Below-floor catch-all: [0, m_bkt0).
56 auto val0 = value_t(m_bkt0);
57 for (size_t idx = 1; idx != m_n_bkts; ++idx, val0 *= value_t(m_ratio))
58 {
59 bkt_val0s[idx] = val0;
60 }
61 /* Last-bucket width such that its expositional max is the next ladder rung -- keeping the geometric
62 * progression uniform through the end. (Beyond-max values land in the last bucket anyway.) */
63 return Histogram_counter{bkt_val0s, bkt_val0s.back() * value_t(m_ratio - 1)};
64}

37void Sparse_shm_object::truncate(bipc::offset_t length)
38{
39 using ::ftruncate;
40 using bipc::interprocess_exception;
41 using bipc::error_info;
42 using bipc::system_error_code;
43
44 /* See our doc header for all relevant background. In short: an overzealous/non-sparse# Base::truncate() =
45 * posix_fallocate(hndl, length), then ftruncate(hndl, length). This truncate() = just the latter; and
46 * commit() performs the equivalent of the former (but re-determines `length` automatically itself).
47 *
48 * # - Whether Base::truncate() is actually overzealous/non-sparse depends on Boost version.
49 * We don't necessarily care; we just do what we advertised. Just pointing out that we internally do the same
50 * stuff as a certain vanilla bipc would, modulo *which* actual method does it. */
51 int rc;
52 do
53 {
54 rc = ftruncate(get_mapping_handle().handle, length);
55 }
56 while ((rc == -1) && (errno == EINTR));
57
58 if (rc == -1)
59 {
60 throw interprocess_exception{error_info{system_error_code()}};
61 }
62}

64void Sparse_shm_object::commit()
65{
66 using ::posix_fallocate;
67 using bipc::interprocess_exception;
68 using bipc::error_info;
69 using bipc::system_error_code;
70
71 /* This is the half of Base::truncate() that this->truncate() omits, made available on demand instead.
72 * posix_fallocate() commits every not-yet-committed page in [0, length): pages already written stay as they are.
73 * It is therefore idempotent and may be invoked at any time after truncate(), including after the object has
74 * been mapped and used. On tmpfs an ENOSPC failure rolls back whatever this call itself had allocated. */
75 bipc::offset_t length;
76 if (!get_size(length)) // Does fstat(); false means it failed.
77 {
78 throw interprocess_exception{error_info{system_error_code()}};
79 }
80 // else
81
82 int err;
83 do
84 {
85 err = posix_fallocate(get_mapping_handle().handle, 0, length);
86 }
87 while (err == EINTR);
88
89 if (err != 0)
90 {
91 throw interprocess_exception{error_info{err}};
92 }
93}

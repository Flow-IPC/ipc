124bool Shared_name::sanitized() const
125{
126 using flow::util::in_closed_range;
127
128 // Keep in sync with sanitize()!
129
130 if (size() > S_MAX_LENGTH)
131 {
132 return false;
133 }
134 // else
135
136 /* Following is self-explanatory given the contract. It could be made much briefer, probably, by using built-in
137 * string functions and/or string_algo/regex/.... However, for nicer perf, we'd rather do only one linear-time scan.
138 * Finally, could other characters be potentially allowed by the various sys calls? Maybe, but we'd rather be safe
139 * than sorry in terms of portability, and also these chars should be aesthetically good as a convention. */
140
141 bool prev_is_sep = false;
142 for (const auto ch : m_raw_name)
143 {
144 bool is_sep = ch == S_SEPARATOR;
145 if ((!is_sep) && (!sanitized_non_sep(ch)))
146 {
147 return false;
148 }
149 // else
150
151 if (is_sep)
152 {
153 if (prev_is_sep)
154 {
155 return false;
156 }
157 // else
158 prev_is_sep = true;
159 }
160 else
161 {
162 prev_is_sep = false; // Might be no-op.
163 }
164 } // for (ch : m_raw_name)
165
166 return true;
167} // Shared_name::sanitized()

169bool Shared_name::sanitize()
170{
171 using std::string;
172 using std::swap;
173
174 // SEPS are the 2 allowed input separator characters. They must be different. Try to keep this compile-time.
175 constexpr char SEPARATOR_ALT = '/';
176 static_assert(S_SEPARATOR != SEPARATOR_ALT,
177 "The real separator and the alt must be different characters.");
178 constexpr const char SEPS[] = { SEPARATOR_ALT, S_SEPARATOR, '\0' };
179
180 /* The following implementation, while self-explanatory, foregoes briefer approaches in favor of better perf.
181 * Keep in sync with sanitized()! */
182
183 if (empty()) // Get rid of degenerate case for simplicity of below.
184 {
185 return true;
186 }
187 // else at least one character to read and (unless it's illegal) write.
188
189 const auto n = size();
190 const auto start_ptr = m_raw_name.cbegin();
191 const auto end_ptr = m_raw_name.cend();
192 string sanitized_str;
193 sanitized_str.reserve(size());
194
195 for (auto ptr = start_ptr; ptr != end_ptr; ++ptr)
196 {
197 /* We cannot predict, at the start, whether we'll ultimately result in a too-long string, because repeating
198 * separators can actually lower the length. However, if we get to this iteration, then we *will* be writing a
199 * character (unless ch is illegal, but then we'd return false anyway); hence we can detect ASAP whether the
200 * output is too long for sure. */
201 if (sanitized_str.size() >= S_MAX_LENGTH)
202 {
203 return false;
204 }
205 // else not too long yet.
206
207 auto ch = *ptr;
208 const bool is_sep = (ch == S_SEPARATOR) || (ch == SEPARATOR_ALT);
209
210 // Perform nearly identical check to sanitized() (reminder: the 2 functions must remain in sync).
211 if ((!is_sep) && (!sanitized_non_sep(ch)))
212 {
213 return false;
214 }
215 // else: Legal character ch; though it may be sanitized by replacing with a different value.
216
217 if (is_sep)
218 {
219 ch = S_SEPARATOR; // Namely, all separator possibilities are normalized to this one.
220
221 /* Since we found a separator, immediately skip past any more adjacent separators by moving ptr
222 * inside this { body }. Set it to the position just before the first non-separator. That ranges between:
223 * not touching ptr at all (i.e., only one separator after all); and setting it to &*(n - 1) (meaning either
224 * we're at end of m_raw_name, or the rest of it is all separators). Note for() will ++ptr right after this. */
225 auto pos = size_t(ptr - start_ptr);
226 if (pos != (n - 1))
227 {
228 pos = m_raw_name.find_first_not_of(SEPS, pos + 1);
229 if (pos == string::npos)
230 {
231 pos = n;
232 }
233 --pos;
234
235 ptr = start_ptr + pos;
236 }
237 } // if (is_sep)
238
239 sanitized_str += ch;
240 } // for (ptr in [start_ptr, end_ptr))
241
242 // Passed gauntlet in terms of whether we can `return true` (we can and must). Finish up.
243 swap(m_raw_name, sanitized_str); // Constant-time. Then sanitized_str is deallocated.
244 return true;
245} // Shared_name::sanitize()

247bool Shared_name::sanitized_non_sep(char ch) // Static.
248{
249 /* std::isalpha/isalnum(s_c<uchar>(ch)) is tempting, but it's affected by the locale; we want none of that.
250 * std::isdigit(s_c<uchar>(ch)) should be fine... but just screw it, man. */
251 return ((ch >= 'A') && (ch <= 'Z')) || ((ch >= 'a') && (ch <= 'z'))
252 || ((ch >= '0') && (ch <= '9'));
253}

255Shared_name build_conventional_non_session_based_shared_name(const Shared_name& resource_type)
256{
257 assert(!resource_type.empty());
258
259 const auto& SENTINEL = Shared_name::S_SENTINEL;
260 const auto& ROOT_MAGIC = Shared_name::S_ROOT_MAGIC;
261
262 Shared_name name;
263 name /= ROOT_MAGIC;
264 name /= resource_type;
265 name /= SENTINEL;
266 name /= SENTINEL;
267 name /= SENTINEL;
268 return name;
269} // build_conventional_non_session_based_shared_name()

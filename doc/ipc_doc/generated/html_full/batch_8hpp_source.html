34/**
35 * Straightforward Msg_batch_in concept implementation (as taken by
36 * API `{Blob|Native_handle}_receiver::async_receive_*_batch()` concept) usable by any receiver
37 * engaging in *emulated batching*.
38 *
39 * @see Native_handle_receiver concept doc header for an explanation of receive-batching (including what
40 * emulated batching is; and how `Msg_resource_t` template-param relates to things).
41 *
42 * That is it features the following APIs as of this writing:
43 *
44 * - For user (including internal user struc::sync_io::Channel): `size_t max_msg_count`-taking ctor,
45 * initialized(), n_used(), full(), clear_used(), prepare_target_payload(), target_payload_size(),
46 * result_payload_blob(), result_payload_hndl() (for Native_handle_receiver only).
47 * - For sync_io::async_receive_batch_emulation() or equivalents: next_target_blob(), next_target_hndl() (for
48 * Native_handle_receiver only), emulate_result().
49 * - Subtlety: While result_payload_blob() and result_payload_hndl(), where arg `idx >= this->n_used()`,
50 * formally results in undefined behavior, Generic_msg_batch_in intentionally relaxes this restriction;
51 * values up to (not including) `max_msg_count` are allowed. The intention here is to allow a `*this`
52 * to be usable even without emulate_result(), or n_used() for that matter. (The
53 * first use-case for this was an internal scenario in struc::sync_io::Channel, when batch-size is set to 1
54 * at compile-time. Then a simplified code-path inside `Channel` stores a "batch" with `max_msg_count == 1`,
55 * then uses one-message-at-a-time receiving to target next_target_blob() and next_target_hndl() and always
56 * leaves `n_used() == 0`.) (A true-batch-handling code-path still relies on a batch being stored in that
57 * data structure but uses actual batched-receiving on it.)
58 *
59 * ### Why are `next_target_*()` and emulate_result() (and `idx >= n_used()`) public? ###
60 * Please note that we have intentionally made the receiver-engine-facing API publicly accessible. The idea is
61 * that conceivably the Flow-IPC user could make use of it, as it is fairly general. As of this writing, however,
62 * only other Flow-IPC internal code uses this sub-API; but that need not be the case. This also applies to
63 * the aforementioned case where `idx >= this->n_used()` to `result_payload_*(idx)`.
64 *
65 * ### Impl notes ###
66 * It's very straightforward. The main thing to keep in mind for context is that Generic_msg_batch_in provides
67 * value only in helping express batch-receives in terms of single-message-receives. There is little to nothing
68 * algorithmic going on; it just provides the expected batch API for the user (in terms of preparing slots and
69 * examining/consuming results) while letting receiver impls straightforwardly record single-message-receives.
70 *
71 * @tparam Msg_resource_t
72 * See Msg_batch_in concept API.
73 * @tparam NO_HNDLS
74 * See Msg_batch_in concept API.
75 */

291template<typename Msg_resource_t, bool NO_HNDLS>
292Generic_msg_batch_in<Msg_resource_t, NO_HNDLS>::Generic_msg_batch_in(size_t max_msg_count) :
293 m_n_used(0)
294{
295 m_batch.reserve(max_msg_count);
296 // It's empty though. Let them set up each guy via prepare_target_payload(-1) x N times.
297 assert((m_batch.capacity() == max_msg_count) && "We use .capacity() to memorize `max_msg_count` for full(), etc.");
298}

300template<typename Msg_resource_t, bool NO_HNDLS>
301void Generic_msg_batch_in<Msg_resource_t, NO_HNDLS>::prepare_target_payload
302 (const util::Blob_mutable& target_blob, Msg_resource&& msg_resource, size_t idx)
303{
304 if (target_payload_size() != 0)
305 {
306 assert((target_payload_size() == target_blob.size())
307 && "Msg_batch_in concept requires the max message size be the same across all slots.");
308 }
309
310 if (idx == size_t(-1))
311 {
312 // Ensure .emplace_back() wouldn't increase .capacity().
313 assert((!initialized())
314 && "At this time the fully-initialized batch size shall be permanently set via ctor; cannot grow.");
315
316 m_batch.emplace_back(target_blob, std::move(msg_resource));
317 }
318 else if (idx < m_batch.size())
319 {
320 auto& mdt = m_batch[idx];
321 mdt.m_buf_seq = target_blob;
322 mdt.m_resource = std::move(msg_resource);
323 }
324 else // if (idx >= m_batch.size()) && not -1
325 {
326 assert(false && "prepare_target_payload(x) must either replace an existing slot x or add slot via `x = -1`.");
327 }
328}

337template<typename Msg_resource_t, bool NO_HNDLS>
338size_t
339 Generic_msg_batch_in<Msg_resource_t, NO_HNDLS>::result_payload_blob(size_t idx, Msg_resource** msg_resource_ptr)
340{
341 assert(idx < m_batch.size());
342 // Note: We intentionally allow `idx >= m_n_used` due to advanced use cases; see our doc header.
343
344 auto& mdt = m_batch[idx];
345 if (msg_resource_ptr)
346 {
347 *msg_resource_ptr = &mdt.m_resource;
348 }
349
350 return mdt.m_result_n_rcvd;
351}

32void Native_handle::close() noexcept
33{
34 if (null()) // As promised no-op in this case.
35 {
36 return;
37 }
38 // else
39
40#ifndef FLOW_OS_LINUX
41 static_assert(false, "Native ::close() tested in Linux only in this context, though anything POSIXy should be OK.");
42#endif
43 ::close(m_native_handle);
44
45 operator=({}); // As promised nullify it.
46
47 /* Discussion of ::close() versus alternatives:
48 *
49 * Firstly, at least in Linux (likely POSIX, but we haven't rigorously looked into it), ::close() does what we want:
50 * the FD (file descriptOR) no longer refers to the file descriptION; so in particular if it's the last FD to have
51 * been referring to it, then the descriptION goes away. So once all FDs (::dup()ed ones, socket-transmitted
52 * ones, original) are thus ::close()d, then the resource can really go away (as opposed to leaking, in addition to
53 * FD(s) leaking). So, at least it's correct/correct enough.
54 *
55 * Secondly, is there an alternative in our style? Yes but no. Yes in that we'd like to use the "portable"
56 * boost.asio for this job. Indeed, probably, making a posix_descriptor S from hndl_or_null.m_native_handle
57 * and calling S.close() would likely do the job. However, posix::descriptor's dtor is protected, so one cannot
58 * really instantiate a posix::descriptor; must instantiate a sub-class: perhaps stream::descriptor which
59 * adds generic read/write ops like write_some() and read_some(). One might be tempted to try
60 * Peer_socket<Protocol_byte_stream> (stream_protocol::socket) or Peer_socket<Protocol_pkt_stream>
61 * (seq_packet_protocol::socket). Does it matter that m_native_handle might not refer to a stream-style
62 * protocol at all (if using stream_protocol::socket or stream::descriptor) or a SEQPACKET-style one (if using
63 * seq_packet_protocol::socket)? One might guess, "no, it's just close it anyway, whatever." Turns out, actually,
64 * no: some kind of machinery in boost.asio (@todo Might be fun/educational to find out what exact machinery)
65 * seems to detect a mismatch at least with {stream|seq_packet}_protocol::socket, leading to a quite-stealthy
66 * failure to actually .close() successfully... and a leak (total failure of this function to fulfill its contract).
67 * So, in closing, no... not that we could find --
68 *
69 * whereas POSIX ::close() gets the job done.
70 *
71 * @todo Maybe revisit for boost.asio education at least. */
72} // Native_handle::close()

79bool Native_handle::is_open() const noexcept
80{
81#ifndef FLOW_OS_LINUX
82 static_assert(false, "F_GETFD is POSIX but tested in Linux only; look into it when porting.");
83#endif
84 return ::fcntl(m_native_handle, F_GETFD) != -1; // (-1 a/k/a null() => EBADF => false, as promised.)
85}

87Native_handle Native_handle::dup(Error_code* err_code) const
88{
89 using boost::system::system_category;
90 using ::fcntl;
91
92 FLOW_ERROR_EXEC_AND_THROW_ON_ERROR(Native_handle, dup, _1);
93 // ^-- Call ourselves and return if err_code is null. If got to present line, err_code is not null.
94
95 if (null())
96 {
97 *err_code = transport::error::Code::S_INVALID_ARGUMENT;
98 return {};
99 }
100 // else
101
102#ifndef FLOW_OS_LINUX
103 static_assert(false, "F_DUPFD_CLOEXEC is POSIX.1-2008 but tested in Linux only; look into it when porting.");
104#endif
105
106 // (Close-on-exec set atomically, so a fork()+exec() racing us cannot leak the new descriptor into the child.)
107 const handle_t result = fcntl(m_native_handle, F_DUPFD_CLOEXEC, 0);
108 if (result == -1)
109 {
110 *err_code = Error_code{errno, system_category()};
111 return {};
112 }
113 // else
114
115 err_code->clear();
116 return Native_handle{result};
117} // Native_handle::dup()

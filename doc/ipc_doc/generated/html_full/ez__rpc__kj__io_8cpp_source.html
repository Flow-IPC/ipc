36Ez_rpc_kj_io::Ptr Ez_rpc_kj_io::this_thread_obj() // Static.
37{
38 auto this_thread_obj_ptr = s_this_thread_obj_observer.lock();
39 if (!this_thread_obj_ptr)
40 {
41 /* Either: (In this thread) We are being called for the first time ever; so the below Ez_rpc_kj_io ctor call has not
42 * been executed, and we've never returned a shared_ptr (Ptr) to it, hence observer is empty -- was not
43 * observing anything yet.
44 * Or: (In this thread) We are being called not the first time; and are observing a shared_ptr (Ptr)
45 * pointing to result of Ez_rpc_kj_io ctor call; but since we returned it, they let that shared_ptr group reach
46 * ref-count zero; so it was deleted.
47 * Either way: Time to make a new one and observe it (and return it). */
48 s_this_thread_obj_observer = this_thread_obj_ptr = Ptr{new Ez_rpc_kj_io}; // Private ctor; cannot use make_shared().
49 }
50 /* else { (In this thread) We are being called not the first time; and are observing a shared_ptr (Ptr)
51 * pointing to result of setupAsyncIo(); and since we returned it, they still have 1+ objects in that group
52 * around; so it exists still. So we can create (and return) a new member of that shared_ptr group. } */
53 return this_thread_obj_ptr;
54}

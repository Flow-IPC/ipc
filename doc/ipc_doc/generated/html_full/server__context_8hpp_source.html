33/**
34 * The mirror of an opposing PEER-state Client_context, as generated via successful Context_server::accept().
35 *
36 * @see ipc::transport::struc::shm::rpc namespace doc header for an overview of Flow-IPC capnp-RPC support, including
37 * how the present class relates to the others there.
38 *
39 * Client_context in PEER state (i.e., on successful `.sync_connect()`) -- therefore also Server_context -- is
40 * a bundling of:
41 * - ipc::session::Session SHM-enabled variant, also in PEER state
42 * - with no passive-channel-open abilities and
43 * - a session-hosed handler that logs about it; and
44 * - a Session_vat_network suitable to create a zero-copy-enabled #Rpc_system.
45 *
46 * In the case of Client_context session() accesses a `Client_session` in PEER state, while our session()
47 * accesses a `Server_session` in PEER state; but in PEER state they are simply a `Session` (how it was started --
48 * client, server -- does not matter).
49 *
50 * All notes in Client_context doc header relating to its semantics after successful Client_context::sync_connect()
51 * apply to a `*this` equally. A `*this` is by definition always in PEER state until destruction.
52 *
53 * @note Certain perf-knobs may be available through `this->vat_network()`. This is, along with the
54 * above-mentioned more basic notes, introduced in the Client_context doc header.
55 *
56 * @internal
57 *
58 * ### Impl notes ###
59 * There was some thought as to maybe making a `Context` with all PEER-state/common parts; have
60 * Server_context essentially equal it; and having Client_context equal it once in PEER state + the other parts
61 * like `.sync_connect()`. However that common part is so simple/short that the trouble involved would probably
62 * be costlier than the limited copy/paste-like action going on. (As of this writing this file is <250 lines
63 * including license header and comments.)
64 *
65 * @endinternal
66 *
67 * @tparam Server_session_t
68 * For a given Context_server instantiation `S`, this is `S::Session_server_obj::Server_session_obj`.
69 */
70template<typename Server_session_t>

72 public flow::log::Log_context,
73 private boost::noncopyable

85 /**
86 * The Session_vat_network concrete type -- and `capnp::VatNetwork` interface impl -- that we produce.
87 * See Session_vat_network docs; but in short, generally, once constructed this guy is used ~identically
88 * to `capnp::TwoPartyVatNetwork`.

90 using Vat_network = typename Session_obj::Vat_network;

92 /**
93 * Movable smart-pointer handle to a `*this`. Spiritually equivalent to `kj::Own<Server_context<...>>`.
94 *
95 * @internal
96 *
97 * Since we are going for a KJ/capnp-RPC-style API with Context_server et al, initially I (ygoldfel) went
98 * with `kj::Own<>` here, as one undoubtedly would if coding KJ/capnp itself. However I was then trying
99 * to do some internally-needed up-casting and having a really annoying time; at one point a `.release()` would
100 * have helped, but `Own` lacks it intentionally. In short I got annoyed and just reverted to the Flow-IPC-ish
101 * choice of single-ownership-ptr. They're used identically (in the mainstream), and if really required one
102 * can always make an `Own` for this after-all... so just, whatevs!
103 */
104 using Ptr = boost::movelib::unique_ptr<Server_context>;

163 * Constructs us in PEER state. Only callable via Server_context_dtl attorney.
164 *

167 * @param logger_ptr
168 * Logger to use for logging subsequently.

173 * to PEER state (session::Server_session::init_handlers() as of this writing).
174 * @param bidir_transport
175 * For Session_vat_network ctor (see its doc header; briefly: pre-connected stream-socket handle,

177 * @param enable_hndl_transport

205 *
206 * Held via shared handle, as #m_network requires. Corollary: `*m_session` may outlive `*this`.
207 * See Session_vat_network doc header, "Object lifetimes." That is also why we warn about `Logger` and `App`
208 * lifetimes in our ctor doc header.

210 Session_ptr m_session;

218template<typename Server_session_t>
219Server_context<Server_session_t>::Server_context(flow::log::Logger* logger_ptr, kj::AsyncIoContext* kj_io,
220 Session_obj&& session, Native_handle&& bidir_transport,
221 bool enable_hndl_transport,
222 bool sans_shm_transport) :
223 flow::log::Log_context(logger_ptr, Log_component::S_RPC),
224 /* We take ownership of Session_obj... and immediately move-from it into a new Session_obj, that one
225 * wrapped in the shared_ptr handle as required by Session_vat_network m_session for its anonying lifetime
226 * reasons (see its class doc header). */
227 m_session(boost::make_shared<Session_obj>(std::move(session))),
228 m_network(get_logger(), kj_io,
229 sans_shm_transport ? Session_ptr{} : Session_ptr{m_session},
230 std::move(bidir_transport),
231 enable_hndl_transport ? Session_vat_network_base::S_N_MAX_INCOMING_FDS : 0)
232 // ^-- Can throw (unlikely).
233{
234 FLOW_LOG_INFO("rpc::Server_ctx [" << *this << "]: Created in PEER state (presumably by Context_server); "
235 "zero-copy-enabled? = [" << (!sans_shm_transport) << "].");
236
237 m_session->init_handlers([logger_ptr, session = m_session.get()](const Error_code& err_code)
238 { on_session_hosed(logger_ptr, *session, err_code); });
239}

250template<typename Server_session_t>
251void Server_context<Server_session_t>::on_session_hosed(flow::log::Logger* logger_ptr, const Session_obj& session,
252 const Error_code& err_code) // Static.
253{
254 FLOW_LOG_SET_CONTEXT(logger_ptr, Log_component::S_RPC); // @todo Maybe use session.get_logger()?
255 FLOW_LOG_INFO("rpc::Server_ctx-created ipc::session::Session [" << session << "]: session-hosed handler fired "
256 "(code [" << err_code << "] [" << err_code.message() << "]). This is likely normal, as the "
257 "opposing process decided to end session; the RPC-system will have detected same, and user "
258 "RPC session should be ended or ending imminently, at which point proper user code shall "
259 "shut-down this Server_ctx which will shut down the Session_vat_network and lastly the "
260 "ipc::session::Session.");
261
262 // Why do we merely log but in no way report this to `*this` user? See Client_context equivalent; same deal here.
263}

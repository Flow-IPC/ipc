73std::string Process_credentials::process_invoked_as(Error_code* err_code) const
74{
75 using boost::lexical_cast;
76 using boost::system::system_category;
77 using fs::path;
78 using fs::ifstream;
79 using std::ios_base;
80 using std::string;
81 // using ::errno; // It's a macro apparently.
82
83 FLOW_ERROR_EXEC_AND_THROW_ON_ERROR(string, process_invoked_as, _1);
84 // ^-- Call ourselves and return if err_code is null. If got to present line, err_code is not null.
85
86#ifndef FLOW_OS_LINUX
87 static_assert(false, "process_invoked_as() depends on Linux /proc semantics.");
88#endif
89
90 path cmd_line_path{"/proc"};
91 cmd_line_path /= lexical_cast<path>(process_id());
92 cmd_line_path /= "cmdline";
93
94 /* @todo Below in a couple places we take errno as the reason for an ifstream failure. That is not guaranteed by the
95 * standard; but in practice both libstdc++ and libc++ (whose filebuf uses fopen()/fread()) leave the libc errno
96 * in place. If errno were somehow 0, we'd emit a success code alongside an empty result. Nicer: set errno = 0
97 * before each op; if still 0 upon failure emit a dedicated util:: error code (cousin of
98 * transport::error::Code::S_MQ_BIPC_MISC_LIBRARY_ERROR) -- which requires a util:: error table. */
99
100 /* This weird "file" (I believe it's really the kernel just pumping out these characters, as we read, as opposed
101 * to anything actually being stored on a drive) is argv[0], argv[1], ..., right in a row, each one NUL-terminated.
102 * So open in binary mode and just read up to NUL or end-of-file (EOF). */
103 ifstream cmd_line_file(cmd_line_path, ios_base::binary);
104 if (!cmd_line_file.good())
105 {
106 *err_code = {errno, system_category()};
107 return {};
108 }
109 // else
110
111 /* From this point on we are quite tolerant; we only emit an error if we see the file in non-.good() state --
112 * but only if it's not end-of-file. Beyond that we make zero assumptions about what might be in here (spaces?
113 * EOF without NUL? whatever!); see background in our doc header impl section. Even an empty return value
114 * is not, in itself, an error (see our contract).
115 *
116 * Detail: We could use .getline(), but I don't want to deal with assumptions of max-length.
117 * We could use string<<, but isspace() has locale implications, and I just don't wanna get into all that.
118 * Reading one char at a time from a buffered stream should be fine perf-wise in this context. */
119 string result;
120 bool done = false;
121 do
122 {
123 const int ch_or_none = cmd_line_file.get();
124 if (!(done = (!cmd_line_file.good()) || (char(ch_or_none) == '\0')))
125 {
126 result += ch_or_none;
127 }
128 // else { We break out of loop. }
129 }
130 while (!done);
131
132 if ((!cmd_line_file.good()) && (!cmd_line_file.eof()))
133 {
134 assert((errno != 0) && "There had to be *some* reason file reading failed, and it wasn't EOF.");
135 *err_code = {errno, system_category()};
136 return {};
137 }
138 // else
139
140 err_code->clear();
141 return result; // Might still be empty (unlikely though)... but as noted -- that's OK by us.
142} // Process_credentials::process_invoked_as()

175std::ostream& operator<<(std::ostream& os, const Process_credentials& val)
176{
177 const auto pid = val.process_id();
178 const auto uid = val.user_id();
179 const auto gid = val.group_id();
180 if ((pid == 0) && (uid == 0) && (gid == 0))
181 {
182 return os << "null";
183 /* Note, we've intentionally not added `operator bool` or `.empty()` (etc.) to Process_credentials as a logical
184 * convention; it's deliberately minimal/low-level; user can worry about exact interpretation of the individual
185 * values. However, for pithy output -- and that thing only -- we've allowed ourselves the convention to represent
186 * an object that is as-if-default-cted as "null" as a string. */
187 }
188 // else
189 return os << "pid[" << pid << "], user[" << uid << ':' << gid << ']';
190}

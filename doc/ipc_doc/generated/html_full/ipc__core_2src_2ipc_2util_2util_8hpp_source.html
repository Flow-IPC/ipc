28/**
29 * Knobs meant to control the cosmetics of printing (`ostream <<`) a bundling of stats/info for some particular
30 * purpose -- whether a snapshot `struct` (e.g., ipc::shm::arena_lend::jemalloc::stat::Arena_info_dump) or a live
31 * accessor bundle (e.g., ipc::session::Info_collector).
32 *
33 * ### Rationale ###
34 * This is an optional convenience thing: We figured various info-dump ops like the above example will tend to
35 * have at least some knobs in common; so standardizing them in a simple `struct` is nice for convenience/reduced
36 * boiler-plate and nice for keeping patterns generic.
37 */
38struct Info_dump_format
39{
40 // Data.
41
42 /**
43 * If `true`: Print everything you've got; if `false`: refrain from producing multi-page output if possible.
44 *
45 * Moreover: if `false` then a snapshot-populating API *is allowed* to save cycles/resources by consulting
46 * this #m_verbose (before `ostream <<` is ever invoked) so as to not compute/save some potentially-verbose
47 * data.
48 *
49 * @warning For this reason it's best for the user to not change `m_verbose` between snapshot-collection
50 * (e.g., ipc::shm::arena_lend::jemalloc::Ipc_arena::info_dump() => `Arena_info_dump`) and output
51 * via `<<` (e.g., of `Arena_info_dump`). Otherwise unnecessary computation/resource use or misleading
52 * output may occur.
53 *
54 * Default: `true`.
55 */
56 bool m_verbose = true;
57
58 /**
59 * If `true`: Use newlines where appropriate for readability (except at the very end of the output); if `false`:
60 * avoid newlines entirely, if at all possible.
61 *
62 * Default: `true`.
63 *
64 * If #m_multiline comes in conflict with #m_verbose then `m_multiline` wins. That is: Observing
65 * `m_multiline == false` is first-priority, whereas observing `m_verbose == true` is secondary/best-effort.
66 *
67 * ### Example of conflict scenario ###
68 * In the SHM-jemalloc module (ipc::shm::arena_lend::jemalloc), #m_verbose = `true` means including jemalloc
69 * memory-manager's own print-stats-dump output, which is pages and pages of tables and readouts (lots of newlines
70 * and other formatting included); `false` means omit it: print only other carefully curated stats but avoid the
71 * jemalloc-dump.
72 *
73 * `m_verbose = true` and `m_multiline = true` is no problem. `m_verbose = false` and `m_multiline = false`:
74 * also no problem. `m_verbose = false` and `m_multiline = true`: ditto. However:
75 *
76 * `m_verbose = true` and `m_multiline = false` presents a conflict: The jemalloc-dump must include newlines;
77 * erasing them would make the tables unreadable; could in some jemalloc-versions use JSON-output, but JSON
78 * is not human-readable in this context; so the bottom line is can't be done. So: `m_multiline = false` wins.
79 * `m_verbose = true` is to be ignored; skip the jemalloc-dump.
80 */
81 bool m_multiline = true;
82}; // struct Info_dump_format

84} // namespace ipc::util::stat

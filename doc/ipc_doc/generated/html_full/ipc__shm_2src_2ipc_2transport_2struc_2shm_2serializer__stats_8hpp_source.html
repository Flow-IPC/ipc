32/**
33 * Cfg type for #Outer_serializer_stats: heap-side outer envelope of a SHM-backed message
34 * (a small SHM-handle wrapper).
35 *
36 * Histograms here are practically degenerate: the outer envelope is always small (~tens of
37 * bytes); a non-trivial bucket tick indicates a misconfiguration of some kind, not normal
38 * operating range. Buckets are sized accordingly: bucket-0 covers up to 512 bytes; bucket-1 is
39 * overflow.
40 */
41struct Outer_serializer_stats_cfg
42{
43 // Constants.
44
45 /// See Serializer_stats::Snd::m_histo_msg_alloc_sz. Degenerate: bucket-0 = [0, 512), bucket-1 = overflow.
46 static constexpr struc::stat::Histo_cfg S_HISTO_SND_MSG_ALLOC_SZ{2, 512, 512};
47
48 /// See Serializer_stats::Snd::m_histo_msg_used_sz. Mirrors alloc-sz; degenerate sanity check.
49 static constexpr struc::stat::Histo_cfg S_HISTO_SND_MSG_USED_SZ{2, 512, 512};
50
51 /// See Serializer_stats::Snd::m_histo_big_leaf_sz. Mirrors alloc-sz; "did the impossible happen" indicator.
52 static constexpr struc::stat::Histo_cfg S_HISTO_SND_BIG_LEAF_SZ{2, 512, 512};
53
54 /**
55 * See Serializer_stats::Rcv::m_histo_msg_used_sz. Mirrors send-side used-sz; rcv-side outer
56 * envelope = single SHM-handle content (~tens of bytes).
57 */
58 static constexpr struc::stat::Histo_cfg S_HISTO_RCV_MSG_USED_SZ{2, 512, 512};
59}; // struct Outer_serializer_stats_cfg

61/**
62 * Cfg type for #Core_serializer_stats: SHM-side payload of a SHM-backed message
63 * (the actual user data, capnp-built into SHM).
64 *
65 * Bucket configs cover the SHM-side allocation regime which -- there being no transport cap on in-SHM
66 * message size -- is open-ended: messages from a few KiB to 100s of MiB are all plausible. Hence
67 * geometric (power-of-2) scales, giving resolution proportional to magnitude across the whole range;
68 * any practical linear scale would instead lump everything past its reach into one bucket.
69 *
70 * Receive-side configs are placeholder values: the receive side, in this SHM-core context, by definition
71 * does not apply and is not touched: on receipt one simply reads what is already in SHM, a/k/a zero-copy.
72 */
73struct Core_serializer_stats_cfg
74{
75 // Constants.
76
77 /**
78 * See Serializer_stats::Snd::m_histo_msg_alloc_sz. Bucket 0 catches below-first-seg-default (8Ki)
79 * sizes; then power-of-2 rungs through 128Mi (so expositional max 256Mi; bigger yet = overflow).
80 */
81 static constexpr struc::stat::Histo_cfg S_HISTO_SND_MSG_ALLOC_SZ{16, 8 * 1024, 0, 2};
82
83 /**
84 * See Serializer_stats::Snd::m_histo_msg_used_sz. Bucket-0 catches tiny content (actionable:
85 * "wrong builder for this size"); then power-of-2 rungs through 128Mi -- coinciding with
86 * #S_HISTO_SND_MSG_ALLOC_SZ's rungs from 8Ki on up, keeping per-bucket alloc-vs-used slack
87 * comparisons meaningful.
88 */
89 static constexpr struc::stat::Histo_cfg S_HISTO_SND_MSG_USED_SZ{19, 1 * 1024, 0, 2};
90
91 /**
92 * See Serializer_stats::Snd::m_histo_big_leaf_sz. Bucket-0 catches "small big-leaf" (below 64Ki);
93 * then power-of-2 rungs through 512Mi (expositional max 1Gi) -- a single giant `Data`/`List` leaf
94 * can be huge in the SHM regime.
95 */
96 static constexpr struc::stat::Histo_cfg S_HISTO_SND_BIG_LEAF_SZ{15, 64 * 1024, 0, 2};
97
98 /**
99 * See Serializer_stats::Rcv::m_histo_msg_used_sz. Placeholder values: this histogram is never
100 * touched for the SHM-msg-inner payload (zero-copy => nothing to allocate in SHM on receipt).
101 */
102 static constexpr struc::stat::Histo_cfg S_HISTO_RCV_MSG_USED_SZ{2, 1, 1};
103}; // struct Core_serializer_stats_cfg

105/**
106 * Empty type template: distinguishing tag for the SHM-msg-outer (heap-side envelope)
107 * `flow::util::stat::Global_stats<>` singleton, parameterized by the SHM-Arena type so each
108 * Arena has its own singleton. Reach the singleton via #Outer_serializer_global_stats.
109 *
110 * For the SHM-msg-inner (SHM-resident user payload), `Arena` itself is used as the tag -- there
111 * is no separate tag. Reach that singleton via #Core_serializer_global_stats.
112 */
113template<typename Arena>
114struct Shm_msg_outer_tag {};

116/**
117 * Bundling of *everything in serializer global-land* -- as filled, for a given SHM-provider `Arena`, by
118 * serializer_info_dump() -- pretty-printable via `ostream <<` in one shot. The bundle:
119 * - #m_heap: pure-heap (non-SHM-backed) user messages; snapshot of the
120 * struc::stat::Heap_serializer_global_stats singleton.
121 * - #m_outer: heap-side outer envelope of each SHM-backed message; snapshot of the
122 * #Outer_serializer_global_stats singleton.
123 * - #m_core: SHM-resident core (the actual user payload) of each SHM-backed message; snapshot of the
124 * #Core_serializer_global_stats singleton. (Its `m_rcv` part is unused/all-zeroes -- see
125 * struc::stat::Serializer_stats doc header -- and is hence skipped when printing `*this`.)
126 *
127 * All 3 members are `struc::stat::Serializer_stats` variants; see that doc header (in particular its
128 * "Key aliases" section) for the full story of what is tracked, by whom, and why.
129 *
130 * Formatting of the `<<` op is controlled by #m_fmt: `m_multiline` is honored; `m_verbose` has no effect
131 * (there is no verbose-only content to gate).
132 */
133struct Serializer_info_dump
134{
135 // Data.
136
137 /// Formatting knobs for printing `*this` via `ostream <<`. See class doc header.
138 util::stat::Info_dump_format m_fmt;
139
140 /// Pure-heap (non-SHM-backed) user messages: snapshot of struc::stat::Heap_serializer_global_stats.
141 struc::stat::Heap_serializer_stats m_heap;
142
143 /// Heap-side outer envelope of SHM-backed messages: snapshot of #Outer_serializer_global_stats.
144 Outer_serializer_stats m_outer;
145
146 /**
147 * SHM-resident core (user payload) of SHM-backed messages: snapshot of #Core_serializer_global_stats.
148 *
149 * Reminder: Its Serializer_stats::m_rcv shall be zero-filled, as it is not applicable: Once a message is
150 * written in (shared) memory, there is no need receipt operation (that would involve further allocation et al)
151 * necessary. Corollary: `ostream << *this` will omit `m_core.m_rcv` in its printout.
152 */
153 Core_serializer_stats m_core;
154}; // struct Serializer_info_dump

160template<typename Arena>
161void serializer_info_dump(Serializer_info_dump* target_info_dump)
162{
163 using flow::util::stat::stats_assign;
164
165 assert(target_info_dump);
166 stats_assign(&target_info_dump->m_heap, struc::stat::Heap_serializer_global_stats::get().stats_default());
167 stats_assign(&target_info_dump->m_outer, Outer_serializer_global_stats<Arena>::get().stats_default());
168 stats_assign(&target_info_dump->m_core, Core_serializer_global_stats<Arena>::get().stats_default());
169}

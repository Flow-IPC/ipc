42Heap_fixed_builder::Heap_fixed_builder(const Config& config) :
43 flow::log::Log_context(config.m_logger_ptr, Log_component::S_TRANSPORT),
44 // (The rounding-down-multiple thing is specified in Heap_fixed_builder_capnp_message_builder ctor doc header.)
45 m_seg_and_frame_sz_cap((config.m_seg_and_frame_sz_cap / sizeof(::capnp::word)) * sizeof(::capnp::word)),
46 // The builder engine launched here. In the future mutators will cause it to allocate in heap on-demand.
47 m_engine(boost::movelib::make_unique<Capnp_heap_engine>
48 (m_seg_and_frame_sz_cap,
49 config.m_frame_prefix_sz, // Do possibly reserve header space before seg 1.
50 0, // Don't reserve headers before each of seg 2, 3, ....
51 true, // Always use the exponential seg size growth, technically...
52 config.m_segment_sz_init, // ...but if this ~equals m_seg_and_frame_sz_cap then effectively don't.
53 config.m_stats)) // Snd-stats target chosen by config (typically by factory); null = no stats.
54{
55 FLOW_LOG_TRACE("Heap_fixed_builder [" << *this << "]: Fixed-size builder started: "
56 "segment (plus framing) size limit [" << m_seg_and_frame_sz_cap << "], "
57 "seg1 min size [" << config.m_segment_sz_init << "]; seg1 frame header "
58 "[" << config.m_frame_prefix_sz << "].");
59 assert(m_seg_and_frame_sz_cap != 0);
60 assert(config.m_segment_sz_init != 0);
61 /* Note: config.m_segment_sz_init may exceed m_seg_and_frame_sz_cap - config.m_frame_prefix_sz; per Config
62 * contract the engine clamps it to that. (E.g., struc::Channel::heap_fixed_builder_config() counts on this:
63 * it forwards the user's segment1_sz estimate as-is, with the cap set to the channel's max-blob-size.) */
64}

86Heap_fixed_builder& Heap_fixed_builder::operator=(Heap_fixed_builder&& src)
87{
88 /* This exists only to log; otherwise it would be `= default`. Replacing #m_engine destroys the
89 * underlying capnp `MessageBuilder`, where snd-stats finalize -- so mirror dtor's TRACE here so
90 * the audit trail is honest on this path too. */
91 if (&src != this)
92 {
93 if (m_engine)
94 {
95 FLOW_LOG_TRACE("Heap_fixed_builder [" << *this << "]: Fixed-size builder being destroyed "
96 "(move-assigned-over).");
97 }
98
99 // The rest is what `= default` would have done.
100 static_cast<flow::log::Log_context&>(*this) = std::move(src);
101 m_seg_and_frame_sz_cap = src.m_seg_and_frame_sz_cap;
102 m_engine = std::move(src.m_engine);
103 }
104
105 return *this;
106}

115void Heap_fixed_builder::emit_serialization(Segment_bufs* target_blobs, util::Blob_mutable* hdr_blob,
116 Split_segments* split_segs,
117 const Session& session, Error_code* err_code) const
118{
119 using util::Blob_const;
120 using util::Blob_mutable;
121
122 if (flow::error::exec_void_and_throw_on_error
123 ([&](Error_code* actual_err_code)
124 { emit_serialization(target_blobs, hdr_blob, split_segs, session, actual_err_code); },
125 err_code, "Heap_fixed_builder::emit_serialization()"))
126 {
127 return;
128 }
129 // If got here: err_code is not null.
130
131 assert(m_engine && "Are you operating on a moved-from `*this`?");
132
133 /* Reminder: concept API says emit_serialization() can be called >1 times. We do allow that in the below.
134 * Keep in mind that mutations can be executed in-between, so whatever caching for perf one imagines has
135 * to not mess up in the face of that. */
136
137 const auto& seg_and_hdr_sz_cap = m_seg_and_frame_sz_cap;
138 target_blobs->clear();
139 if (split_segs)
140 {
141 split_segs->clear();
142 }
143
144 Segment_ptrs blob_ptrs;
145 m_engine->emit_segment_blobs(&blob_ptrs);
146
147 const auto n_blobs = blob_ptrs.size();
148 assert((n_blobs != 0) && "Empty serialization? What went wrong?");
149 auto n_blobs_with_subsegs = n_blobs; // Small optimization involves this....
150 target_blobs->reserve(n_blobs); // No splits (which is typical) means this optimization is sufficient.
151
152 for (size_t idx = 0; idx != n_blobs; ++idx)
153 {
154 auto& blob = *(blob_ptrs[idx]);
155 const auto buf = blob.mutable_buffer();
156 const auto seg_sz = buf.size();
157
158 /* Header area precedes seg 1 (which must exist); no other headers. Reminder: our workhorse m_engine is
159 * required to allocate the seg and also enough header bytes preceding it. So we just recognize the fruits of
160 * its labor now; and do the post-processing splitting if needed/requested. */
161
162 size_t hdr_sz_or_0;
163 if (idx == 0)
164 {
165 hdr_sz_or_0 = blob.start();
166 *hdr_blob = Blob_mutable{static_cast<void*>(blob.begin() - hdr_sz_or_0), hdr_sz_or_0};
167 }
168 else
169 {
170 hdr_sz_or_0 = 0;
171 }
172
173 FLOW_LOG_TRACE("Heap_fixed_builder [" << *this << "]: "
174 "Serialization segment [" << idx << "] (0-based, of [" << n_blobs << "], 1-based): "
175 "Heap buffer @[" << buf.data() << "] sized [" << seg_sz << "]; "
176 "with header/prefix (if any) sized [" << hdr_sz_or_0 << "].");
177
178 target_blobs->emplace_back(buf); // In case of split below its .size() shall be "corrected."
179
180 // Check for the one error (or special) condition: a too-large segment.
181 if ((seg_sz + hdr_sz_or_0) > seg_and_hdr_sz_cap)
182 {
183 if (!split_segs)
184 {
185 FLOW_LOG_WARNING("Heap_fixed_builder [" << *this << "]: "
186 "Serialization segment [" << idx << "] (0-based, of [" << n_blobs << "], 1-based): "
187 "Heap buffer @[" << buf.data() << "] "
188 "sized [" << seg_sz << "] plus frame header [" << hdr_sz_or_0 << "]: exceeded limit "
189 "[" << seg_and_hdr_sz_cap << "]. Seg-split not requested (why not? bug?) so: "
190 "Emitting error; will not return serialization.");
191 *err_code = error::Code::S_INTERNAL_ERROR_SERIALIZE_LEAF_TOO_BIG;
192 return;
193 }
194 // else if (split_segs):
195
196 /* If seg is too big, split it up into the following, in order:
197 * - seg itself: Set its size to <cap minus hdr_sz_or_0> instead of the too-big size.
198 * - [n_full_cont_subsegs times]: "Whole" (cap-sized) continuation sub-segments.
199 * - [Only if a "partial" (<cap-sized) # of bytes is left over after that]: Final sub-segment.
200 * n_cont_subsegs = n_full_cont_subsegs from bullet two, plus 1 if bullet three applies.
201 *
202 * Note we aren't actually copying any of the contents of the segment, just figuring out the locations
203 * and sizes of the sub-segments within it. (After IPC-transmission, reassembly will involve copying, as
204 * the sub-segments shall be scattered in receiver's RAM.) */
205
206 // Finalize the original segment (now sub-seg 1).
207 auto last_subseg_ptr = static_cast<uint8_t*>(buf.data());
208 auto last_subseg_sz = static_cast<size_t>(seg_and_hdr_sz_cap - hdr_sz_or_0);
209 target_blobs->back() = Blob_const{last_subseg_ptr, last_subseg_sz};
210 const size_t seg_minus_subseg1_sz = seg_sz - last_subseg_sz;
211 // seg_minus_subseg1_sz = remaining # of bytes from original segment (all but sub-seg 1).
212
213 // Bullet two above: This many full sub-segments (possibly 0).
214 size_t n_full_cont_subsegs = seg_minus_subseg1_sz / seg_and_hdr_sz_cap;
215 size_t n_cont_subsegs = n_full_cont_subsegs;
216
217 // Bullet three above: Possibly 1 more sub-segment.
218 const size_t partial_cont_subseg_sz_or_0 = seg_minus_subseg1_sz % seg_and_hdr_sz_cap;
219 (partial_cont_subseg_sz_or_0 != 0) && (++n_cont_subsegs);
220
221 // Little additional optimization to try minimizing shifting/reallocs due to the "unexpected" split-related segs.
222 target_blobs->reserve(n_blobs_with_subsegs += n_cont_subsegs);
223
224 // Okay, it's planned-out/accounted; update the actual out-args target_blobs and split_segs.
225
226 for (; n_full_cont_subsegs != 0; --n_full_cont_subsegs)
227 {
228 last_subseg_ptr += last_subseg_sz;
229 last_subseg_sz = seg_and_hdr_sz_cap;
230 target_blobs->emplace_back(last_subseg_ptr, last_subseg_sz);
231 }
232
233 if (partial_cont_subseg_sz_or_0 != 0)
234 {
235 last_subseg_ptr += last_subseg_sz;
236 target_blobs->emplace_back(last_subseg_ptr, partial_cont_subseg_sz_or_0);
237 }
238
239 split_segs->emplace_back(Split_segment{ target_blobs->size() - n_cont_subsegs - 1,
240 n_cont_subsegs });
241 FLOW_LOG_TRACE("Heap_fixed_builder [" << *this << "]: "
242 "Serialization segment [" << idx << "] (0-based, of [" << n_blobs << "], 1-based): "
243 "Heap buffer @[" << buf.data() << "] "
244 "sized [" << seg_sz << "] plus frame header [" << hdr_sz_or_0 << "]: exceeded limit "
245 "[" << seg_and_hdr_sz_cap << "]. Seg-split requested; therefore split into "
246 "[" << (n_cont_subsegs + 1) << "] sub-segments sized "
247 "init[" << (seg_and_hdr_sz_cap - hdr_sz_or_0) << "], N x full[" << seg_and_hdr_sz_cap << "], "
248 "partial[" << partial_cont_subseg_sz_or_0 << "].");
249 } // if ((seg_sz + hdr_sz_or_0) > seg_and_hdr_sz_cap)
250 // else if ((seg_sz + hdr_sz_or_0) <= seg_and_hdr_sz_cap) { Segment need not be split. }
251 } // for (blob : blob_ptrs)
252 // Got here: no error. Done and done.
253 err_code->clear();
254} // Heap_fixed_builder::emit_serialization()

281Heap_reader::~Heap_reader()
282{
283 using flow::util::stat::fetch_sub;
284
285 if ((!m_stats) || (!m_engine)) // I.e., if (nothing to do except log):
286 {
287 FLOW_LOG_TRACE("Heap_reader [" << *this << "]: Reader being destroyed including: "
288 "serialization containing [" << m_serialization_segments.size() << "] segments; "
289 << (m_engine ? "no stats kept." : "no deserialization yet attempted."));
290 /* If !m_engine: deserialization() never succeeded; nothing was incremented; nothing to decrement.
291 * If !m_stats: stats not kept at all. Either way, skip the decrements. */
292 return;
293 }
294 // else
295
296 // This *might* be perf-pricey, to some extent, so definitely be careful with the log level.
297 FLOW_LOG_TRACE("Heap_reader [" << *this << "]: Reader being destroyed including: "
298 "serialization containing [" << m_serialization_segments.size() << "] segments; "
299 "last deserialization reports [" << (m_engine->sizeInWords() * sizeof(::capnp::word)) << "] "
300 "input heap space total.");
301
302 /* Decrement gauges that deserialization() incremented. Underflow assertions ride on the RMW's
303 * pre-change return value. */
304#ifndef NDEBUG
305 const auto prev_msgs_outstanding =
306#endif
307 fetch_sub(&m_stats->m_msgs_outstanding, 1);
308 assert((prev_msgs_outstanding >= 1)
309 && "Bug: rcv-stats msgs_outstanding gauge underflow at Heap_reader dtor.");
310
311#ifndef NDEBUG
312 const auto prev_alloc_outstanding_sz =
313#endif
314 fetch_sub(&m_stats->m_alloc_outstanding_sz, m_alloc_sz);
315 assert((prev_alloc_outstanding_sz >= m_alloc_sz)
316 && "Bug: rcv-stats alloc-bytes_outstanding gauge underflow at Heap_reader dtor.");
317
318#ifndef NDEBUG
319 const auto prev_used_outstanding_sz =
320#endif
321 fetch_sub(&m_stats->m_used_outstanding_sz, m_used_sz);
322 assert((prev_used_outstanding_sz >= m_used_sz)
323 && "Bug: rcv-stats used-bytes_outstanding gauge underflow at Heap_reader dtor.");
324} // Heap_reader::~Heap_reader()

326void Heap_reader::add_serialization_segment(Segment_blob_in&& blob)
327{
328 using flow::util::Blob;
329 using Ptr = const void*;
330
331 static_assert(std::is_nothrow_move_constructible_v<Segment_blob_in>,
332 "Segment_blob_in (as of this writing Basic_blob) move-ct must be noexcept for vector to move them "
333 "instead of copying on realloc; this is critical not just for performance, but because callers "
334 "may routinely rely on a given segment not moving in memory across add_serialization_segment()s.");
335
336 FLOW_LOG_TRACE("Heap_reader [" << *this << "]: "
337 "Serialization segment [" << m_serialization_segments.size() << "] (0-based) added/owned: "
338 "Heap buffer @[" << static_cast<Ptr>(blob.const_data()) << "] "
339 "max-sized [" << blob.capacity() << "], "
340 "actual serialization sized [" << blob.size() << "].");
341
342 /* Out of paranoia ensure seg1's memory never moves; the static_assert() above should ensure it already but still....
343 * (None of the others should move either, but we'll just check it for that particular one.) */
344#ifndef NDEBUG
345 Ptr seg1_ptr = nullptr;
346 m_serialization_segments.empty()
347 || (seg1_ptr = static_cast<Ptr>(m_serialization_segments.front().const_data()));
348#endif
349
350 m_serialization_segments.emplace_back(std::move(blob));
351 assert(((!seg1_ptr) || (seg1_ptr == static_cast<Ptr>(m_serialization_segments.front().const_data())))
352 && "Bug: vector realloc moved a segment blob's buffer, presumably via unintended Basic_blob copy; "
353 "make sure Basic_blob/Segment_blob_in is coded properly for by-value storage in containers.");
354}

64 const std::shared_ptr<Memory_manager>& memory_manager,
65 Shared_name&& pool_name_base,

72 Thread_lcl_obj_db_admin<Ipc_arena>::this_thread_piggy_scan(); // Opportunistic!

85void Ipc_arena::destroy()
86{
87 using ipc::shm::arena_lend::detail::Thread_lcl_obj_db_admin;
88 using ipc::shm::arena_lend::detail::Thread_lcl_obj_db_client;
89 using flow::log::Log_context_mt;
90 using flow::util::ostream_op_string;
91
92 using std::shared_ptr;
93 namespace str_algo = boost::algorithm;
94 using boost::adaptors::transformed;
95
96 /* Reminder: This is not some normal method executed normally. Explanation:
97 *
98 * When is a *this destroyed? Answer: Once the shared_ptr ref-count for the group returned
99 * by create() factory reaches 0; and that occurs not merely when the user's outer shared_ptr copies/etc.
100 * all disappear; but also (possibly among more things) all the shared_ptr<T>s returned by e.g.
101 * construct<T>() also disappear: we intentionally capture a copy of this->shared_from_this() in the
102 * disposer for each such shared_ptr<T>. So they have to drop `*this` *and* all the objects
103 * this->construct<T>()ed from it. That is when, by default, the default deleter runs which simply
104 * performs `delete this`. In our case, though, this may need to be deferred for a few different activities
105 * to complete first.
106 *
107 * At the current stage. X=shared_ptr<Ipc_arena> handle reaching ref-count 0 implies 3 key things
108 * (indeed that ref-count is 0 now):
109 * - `Shm_session`s to which X was added have all been destroyed (which by the way itself implies that
110 * all SHM-object-shared_ptr-handles borrow_object()ed through them have also reached their ref-count=0).
111 * - User has dropped X and any copies.
112 * - User has dropped any SHM-object-shared_ptr-handles construct()ed through *this. <-- ATTN
113 * A copy of X was placed into the shared_ptr disposer for each one; but now they are all gone.
114 *
115 * The last one (ATTN) is somewhat subtle: It means in this process there are no more references to every
116 * object Y acquired as `Y = this->construct<T>(...)`. It however does not mean every such object Y has
117 * actually been deallocated (from SHM!). There are at least two reasons for this, although one really
118 * subsumes the other; but we mention both in the interests of education/explication. The lesser reason is
119 * that even if deallocation always occurred the moment it was warranted (no more users in either the lender
120 * or the borrower remain), we might have lost contact with an opposing Shm_session prematurely -- e.g.,
121 * just simply an opposing-process crash -- so the use-count decrements w/r/t objects borrow_object()ed by
122 * that side never happened and might never happen; so a deallocation hasn't been warranted for them
123 * (use-count must reach 0).
124 *
125 * But the greater/subsuming reason is that, assuming no crashing or anything weird, deallocations do *not*
126 * occur instantly upon a given object Y's use-count reaching 0. In fact only the Thread_lcl_obj_db_admin
127 * per-thread object belonging to the thread that originally allocated Y -- or its degraded-mode thread
128 * (see Thread_lcl_obj_db_admin; but we digress) -- is the one, exclusively, performing scans of
129 * Y's use-count. It can only do that opportunistically -- when a user calls into any of our APIs, and we
130 * opportunistically perform Thread_lcl_obj_db_admin::piggy_scan(). We don't control the event loop of
131 * that thread -- it's the user's thread! -- so this will only happen when it will happen; we can't, like,
132 * .post() a thing that will do it.
133 *
134 * Naturally, deleting Y cannot occur once ~Ipc_arena() dtor executes; the jemalloc-arenas would be
135 * potentially eliminated immediately (or a bit later depending on Thread_cache situation). So then we
136 * have a few options ostensibly. 1, we could just... not delete Y. After all if the arena holding it
137 * will be destroyed wholesale, who cares? Well, no: User did nothing wrong whatsoever, and there haven't
138 * been any IPC-errors necessarily; they at least expect the dtor of type *Y to run. Of course we could
139 * just execute dtor of the type *Y, but that brings us back essentially to the original problem; if the
140 * type is an STL-compliant thing that was using our Stateless_allocator, then invoking that dtor will still
141 * deallocate stuff, just not the outer object pointed-at by Y itself.
142 *
143 * Well then, 2, why *don't* we just delete Y right now, from *this* thread, while we have the instruction
144 * pointer pointing into us?! Answer: Ignoring the perf issues with that (it foregoes thread caching),
145 * consider that now it's possible for another thread to be completely validly right now detecting that
146 * the use-count for the Y pointee *is* zero -- it might be getting around to deleting Y right now itself.
147 * So we'd have to synchronize, using a mutex or something both in here and in
148 * Thread_lcl_obj_db_admin::piggy_scan(). This blows up the simplicity and lock-freeness of the existing
149 * design; though given that only 2 threads can ever contend for this mutex, it isn't the most awful idea
150 * actually. Still, I (ygoldfel) am not a fan at least.
151 *
152 * So that leaves 3: Since we must wait for the proper thread to delete Y, and this will happen at the
153 * earlier of either the user calling into *a* SHM-jemalloc API from that thread, or that thread exiting
154 * normally, why not delay the actual destruction of *this until then? So that's why we'll chain the
155 * "retirement" of a *this as follows:
156 * - create()-returned shared_ptr<Ipc_arena> reaches ref-count 0 =>
157 * - our custom disposer runs... but doesn't delete the Ipc_arena; instead =>
158 * - it creates a new shared_ptr<Ipc_arena> Z with a different custom disposer; and
159 * - calls static Thread_lcl_obj_db_admin::forgetting_shm_arena(Z) which memorizes moved()d Z
160 * inside itself; and
161 * - sets up a cross-thread assignment to per-thread Thread_lcl_obj_db_admin thread-local objects,
162 * specifically to (1) opportunistically forget all objects originated-from arena Z; and (2) once done:
163 * - forget shared_ptr Z itself; which causes Z=shared_ptr<Ipc_arena> to reach ref-count 0 =>
164 * - our 2nd custom disposer runs... but still doesn't delete the Ipc_arena; instead =>
165 * - ...continue procedure. Namely execute helper destroy_on_obj_db_forgot_us(); see further commentary there.
166 *
167 * Worth noting, even if it's not strictly within our purview here:
168 * - Due to the "lesser reason" a few paragraphs back,
169 * Thread_lcl_obj_db_admin::forgetting_shm_arena() shouldn't be looking for use-count=0 objects
170 * like a normal this_thread_piggy_scan(): due to e.g. an opposing-side crash, that might never be reached --
171 * but an Ipc_arena going away by contract means we consider that moot. So it should get rid of
172 * everything to do with our `m_arena` ID(s), whether or not use-count 0 is in effect for a given thing.
173 * - Specifically, take object Y this->construct()ed by thread T1; all such objects must be deleted
174 * before Thread_lcl_obj_db_admin will be able to move on-to Ipc_arena::destroy_on_obj_db_forgot_us().
175 * When will this occur?
176 * - If thread T1 is still live: The earlier of:
177 * - T1 exits (~Thread_lcl_obj_db_admin() dtor runs for that thread-local obj-DB);
178 * - In T1, user calls into any SHM-jemalloc API.
179 * - If thread T1 has exited (then it would have launched degraded-mode thread T1' of its own):
180 * - Next time T1' periodically wakes up (as of this writing every 100msec). */
181
182 Thread_lcl_obj_db_admin<Ipc_arena>::this_thread_piggy_scan(); // Opportunistic!
183
184 // Log ~final info/stats.
185 FLOW_LOG_INFO("Ipc_arena [" << this << "] handle has reached ref-count=0. "
186 "Before totally destroying it, we issue order to the cross-thread obj-DB admin/client-modules "
187 "to forget this arena; admin-part shall occur potentially asynchronously, piggybacked onto "
188 "user API calls and/or relevant threads exiting; once all have finished this "
189 "forget-arena op, the last one to do so shall continue to destroy the Ipc_arena.");
190 FLOW_LOG_INFO("Ipc_arena [" << this << "] shutdown: jemalloc-arena-ID list: " // Very useful diagnostic.
191 "[" << str_algo::join(m_arenas | transformed([](auto arena_id) -> auto
192 { return ostream_op_string(arena_id); }),
193 ", ") << "]."); // BTW: reminder: they're in sorted order in m_arenas.
194 {
195 /* Don't need the many-pages jemalloc-dump (verbose=false). multiline=true is fine however.
196 * Avoid unneeded info_dump() by placing it inside log-macro. */
197 Info_dump dump;
198 dump.m_fmt.m_verbose = false;
199 FLOW_LOG_INFO("Ipc_arena [" << this << "] shutdown: "
200 "~Final state (includes ~final this-arena + ~current global):\n"
201 << (info_dump(&dump), dump) << '.'); // Note: no newline at end of info_dump.
202 }
203
204 /* This is also required but is orthogonal to the much-discussed _admin-oriented stuff and lacks its complexities.
205 * Here it's entirely about cleanliness. At any rate: it's documented as being required, so do it synchronously. */
206 Thread_lcl_obj_db_client<Ipc_arena>::forgetting_shm_arena(get_id());
207 // Now back to complicated _admin stuff.
208
209 /* Logging possibilities are about to get async-weird; and the first thing to potentially trigger asyncness
210 * if the following _admin::forgetting_shm_arena() call. So it's time to lock things down.
211 *
212 * Firstly save this; get_logger() is safe until the current method destroy() returns (to reiterate: it's on
213 * the user to ensure resources given to Ipc_arena stay alive while Ipc_arena's `shared_ptr` group is alive --
214 * which is still the case at the moment; we are in the disposer for that smart-pointer group). So just make a
215 * silly Log_contex_mt whose contents are valid and cannot change until the present method returns.
216 *
217 * We can give it to forgetting_shm_arena() et al; and for sync logging we can use it ourselves. */
218 const Log_context_mt log_ctx_sync{get_logger(), get_log_component()};
219 FLOW_LOG_SET_LOCKED_CONTEXT(&log_ctx_sync);
220 /* Secondly, before asyncness might begin, ensure that any "regular" *this logging code (that uses get_logger()
221 * such as FLOW_LOG_INFO/TRACE/WARNING()) sees null get_logger(). At this stage it is thread-safe: user cannot
222 * be messing with *this while its original handle has ref-count 0; no construct()ed objects remain, so no
223 * concurrent allocs/pool creations/etc. can occur; and we haven't kicked off anything async/concurrent yet.
224 * The one exception: obj-db-deferred deallocations may run -- via an obj-db scan in some thread -- until the
225 * _admin::forgetting_shm_arena() barrier below; hence deallocate() specifically must not (and does not) touch
226 * get_logger() (see comment there). */
227 set_logger(nullptr);
228 // Now, any "regular" logging => no-op; but _LOCKED() logging shall work synchronously.
229
230 // Kick off the (potential) async-craziness.
231 const bool obj_db_forgot_us_synchronously
232 = Thread_lcl_obj_db_admin<Ipc_arena>
233 ::forgetting_shm_arena(get_id(),
234 get_logger(), // For synchronous logging only (get_logger() is safe while in destroy()).
235 [this_arena_on_death_row_ptr = this]
236 (const Log_context_mt* log_ctx) mutable
237 {
238 Thread_lcl_obj_db_admin<Ipc_arena>::this_thread_piggy_scan(); // Opportunistic!
239
240 /* Careful. If this is executing, then forgetting_shm_arena() did *not* complete its stuff synchronously,
241 * and we're not executing within Ipc_arena::destroy() but rather form another thread.
242 * User is allowed to let that *get_logger() be destroyed: they don't know the Ipc_arena
243 * is still hanging around due to Thread_lcl_obj_db_admin (TLODBA) technicalities. To help us with that
244 * TLODBA::forgetting_shm_arena() gives us its Log_context_mt* (Logger*
245 * therein could still be null, but the point is it's something
246 * the user is at least indirectly in charge of ensuring exists while TLODBA is in action). So
247 * we log the following message using that Logger*, and we set the Ipc_arena's Logger to that.
248 * Then inside the .reset() below this->destroy_on_obj_db_forgot_us() will be executing -- synchronously --
249 * with a valid Logger*. At least, the synchronous part of it (but not our problem here). */
250
251 FLOW_LOG_SET_LOCKED_CONTEXT(log_ctx);
252 FLOW_LOG_INFO_LOCKED("Ipc_arena death-row disposer executing for one of the per-thread admins, in thread "
253 "different from the one in which the shared_ptr<Ipc_arena> disposer initiated shutdown; "
254 "according to that last obj-DB admin: arena is safe to continue to destroy.");
255
256 this_arena_on_death_row_ptr->destroy_on_obj_db_forgot_us(log_ctx);
257 }); // Thread_lcl_obj_db_admin::forgetting_shm_arena()
258
259 if (!obj_db_forgot_us_synchronously)
260 {
261 return; // The above lambda is memorized by _admin and will execute from another thread. Done here.
262 }
263 /* else: The above lambda is forgotten: it is safe for us to proceed synchronously.
264 *
265 * We do basically the same thing, but as for logging: we are still synchronously in destroy(); get_logger()
266 * is still safe; use the Log_context_mt we've set-up for that purpose. (Our own log-call here uses it too.) */
267
268 FLOW_LOG_INFO_LOCKED("shared_ptr<Ipc_arena> disposer initiated shutdown; obj-DB reports that no async "
269 "arena-forgetting work remains, so we can immediately/synchronously continue to "
270 "destroy arena.");
271 destroy_on_obj_db_forgot_us(&log_ctx_sync);
272
273 /* Careful! It's very much possible this->destroy() *also* synchronously is able to finish
274 * what it must -- and *right now here* *this is gone. So no more this-> statements! */
275} // Ipc_arena::destroy()

277void Ipc_arena::destroy_on_obj_db_forgot_us(const flow::log::Log_context_mt* log_ctx)
278{
279 using Owner_shm_pool_repository = ipc::shm::arena_lend::detail::Owner_shm_pool_repository<Ipc_arena>;
280 using flow::log::Log_context_mt;
281 using std::shared_ptr;
282
283 assert((!get_logger()) && "Regular get_logger()-based logging should have been disabled earlier by "
284 "destroy(); it's *log_ctx or bust from now on.");
285
286 /* Per contract, any synchronous logging must go through *log_ctx. (In practice, it is either
287 * the regular get_logger() in-effect when destroy() was called, and we are still in destroy(); or this
288 * is happening asynchronously/past destroy(), and *log_ctx is some Thread_lcl_obj_db_admin's.) */
289 FLOW_LOG_SET_LOCKED_CONTEXT(log_ctx);
290
291 /* What exactly happens from this point is in my (ygoldfel) opinion notoriously difficult to grasp from
292 * reading the code.
293 *
294 * At risk of redundancy (versus destroy*() doc header(s) and further comments below) here is an
295 * annotated recap to help navigate this touchy area.
296 *
297 * -# m_destroy_started = true. (Effect below.)
298 * -# For each of m_shm_pools, for each of m_listeners, .notify_removed_shm_pool().
299 * - If N = # of `Shm_session`s we are lent-through: There are N listeners through which this does:
300 * Shm_session::remove_lender_shm_pool().
301 * - As of this writing: no-op. Might change to inform opposing process of pool-removal.
302 * -# For each of m_shm_pools: Owner_shm_pool_repository::get_instance().erase(pool_id).
303 * -# For each jemalloc-arena A (as of this writing: 1 of them):
304 * jemalloc::Memory_manager::destroy_arena(A)
305 * -# (The following occurs [a]synchronously in 1+ threads including this one.
306 * It is potentially async, because via jemalloc::Thread_cache::destroy_arena_safely(), this must
307 * occur via piggy-backing in 0+ other threads with jemalloc tcaches. If only this thread has tcache(s),
308 * then it is synchronous.)
309 * For each arena A:
310 * - At least some (presumably all?) jemalloc-extents (vaddr-areas) are deinitialized/let-go-of.
311 * This triggers, via still-registered jemalloc-extent-hooks:
312 * pool-removal method(s) execute, cleaning up book-keeping; closing SHM-pool handles; deleting
313 * SHM-pool names from file-system (SHM-RAM returned to OS once all of that, plus SHM-pool handle freeing,
314 * also occurs in borrower(s), at latest on Shm_session destruction that
315 * might follow soon/already happened/happening now (but not necessarily, depending on user behavior; might
316 * be later).
317 * - Ipc_arena::on_shm_pool_removed() is called. No-op: m_destroy_started is true.
318 * We already did what this would do, in step 3 above. (Trying it again is not harmless; in particular
319 * various pools' base vaddrs are no longer in Owner_shm_pool_repository -- step 3 did it -- so it'd
320 * try to remove non-existent things and crash.)
321 * -# (This [a]sync-follows the last step of the destruction of the last arena A to be destroyed.)
322 * `delete this`. `*this` Ipc_arena is destroyed. Destructor is essentially no-op other than freeing RAM;
323 * The per-pool SHM-pool-handles should have been closed via arena destruction.
324 * Note: This last step is deferred, because the registered jemalloc-extents (their required bookkeeping) are
325 * in *this.
326 *
327 * We will explain hopefully each step again, but without the above tactical recap it has proven difficult
328 * to see the forest for the trees in the past. */
329
330 /* Prevents at least Owner_shm_pool_repository erase and listener notification from happening
331 * below in the arena destruction (on_shm_pool_removed() early-returns when m_destroy_started). */
332 m_destroy_started.store(true);
333
334 /* Run listeners that would presumably otherwise run in arena destruction below as triggered
335 * via jemalloc extent-hooks. We've prevented that now and choose to do so all in one group from this one thread
336 * and with no possibility of skipping any SHM-pools (if jemalloc for whatever reason doesn't trigger that).
337 * We feel this involves less entropy overall.
338 *
339 * Then deregister from Owner_shm_pool_repository (after listeners, for consistency with on_shm_pool_removed()). */
340 {
341 Lock lock{m_shm_pools_and_listeners_mutex}; // Probably not necessary by now but just in case....
342
343 FLOW_LOG_TRACE_LOCKED("Notifying listeners of [" << m_shm_pools.size() << "] shared memory pool removals; these "
344 "will not run via jemalloc extent-hooks anymore (though extent-hooks will execute soon; but "
345 "listeners will be skipped).");
346 for (const auto& cur_shm_pool : m_shm_pools)
347 {
348 for (auto cur_listener : m_listeners)
349 {
350 cur_listener->notify_removed_shm_pool(cur_shm_pool);
351 /* Logging safety mental check: Formally speaking the logging by whatever that triggers is that thing's
352 * responsibility; we do not log there -- just invoke their virtual handler.
353 *
354 * Just in case let's think what "that thing" might be: As of this writing `Shm_session`s register
355 * themselves for these updates; that's it. Each Shm_session we are lent-through (.lend_arena(this)
356 * essentially) does so. That said, as of this writing, the Ipc_arena would not be destroyed until
357 * those `Shm_session`s have all dropped their shared_ptr refs to *this; and that (again... as of this
358 * writing) only happens when the `Shm_session`s go away which would unregister from m_listeners
359 * (but so would a hypothetical .unlend_arena() which does not exist as of this writing). So really
360 * this should not even be executing.
361 *
362 * If other registering listeners exist, it might be a different story, but even then "their" logging is
363 * "their" problem. */
364 }
365 } // for (cur_shm_pool : m_shm_pools)
366
367 for (const auto& cur_shm_pool : m_shm_pools) // Pool-repository deregistration.
368 {
369 Owner_shm_pool_repository::get_instance().erase(cur_shm_pool->get_id());
370 /* Logging safety mental check: Owner_shm_pool_repository has its own logging setup as a singleton; it
371 * (and subordinate things like pool-lookup thread-local machinery) are responsible for maintaining
372 * safe Logger*s and all. */
373 }
374 } // Lock lock{m_shm_pools_and_listeners_mutex}
375
376 /* The rest: destroy jemalloc arenas. See the big comment recap above for context.
377 *
378 * Please see destroy() for key background; then come back here and keep reading.
379 * What does our dtor need to do? Just this: For each ID in m_arenas, just:
380 * get_jemalloc_memory_manager()->destroy_arena(arena_id);
381 *
382 * It's actually important to know what it does. Consider one arena
383 * (it's the same for each one; plus realistically we just have one in practice as of this writing, per *this).
384 * It won't just blow-up the vaddr areas a/k/a extents in the arena, and that's it. That'd be nice arguably but...
385 * actually it'll in civilized fashion, like, unmap/remove extents and things of that nature -- which in fact
386 * will trigger the relevant extent hooks (see ctor), most notably remove_shm_pool_handler() which calls
387 * remove_shm_pool() which calls super-class Owner_shm_pool_collection::remove_range_and_pool_if_empty()
388 * which unmaps the extent and closes the SHM-pool (SHM-object) handle and even unlinks the pool name
389 * (it disappears from file system, and once all handles across the system are also closed, the RAM is returned
390 * fully).
391 *
392 * No problem though, right? Just call ->destroy_arena(). Sadly -- see its doc header -- that doesn't
393 * simply do the mallctl("...destroy") call; and that's because it potentially cannot. You can read
394 * its doc header and/or that for Thread_cache::destroy_arena_safely() (which is what it calls really), but
395 * long story short: It is mandatory to destroy (really merely flush, but at this point why not destroy too)
396 * every tcache that has ever been used in 1+ allocations against arena_id.
397 *
398 * Well, who cares? Fine then, let ->destroy_arena() do it, however long it takes asynchronously; why should we
399 * care? Sigh: nope. By that time, if we let our dtor run *now*, the extent hooks we'd registered
400 * will refer to a non-existent *this; so that'll crash probably; and anyway the extents (pools) might be around
401 * and mapped, but our Shm_pool objects, etc., are gone.
402 *
403 * Once that becomes clear the solution isn't that hard; we "just" can't let *this die yet. ->destroy_arena()
404 * takes an optional functor to execute right after it's actually safe to destroy the arena. So we can capture
405 * `this` and delete it in the functor. destroy(), which led to us, used a similar technique, but it couldn't
406 * delete `this` once done with is charge; so it called us. We are the last thing though and can delete `this`
407 * once done. */
408
409 shared_ptr<Ipc_arena> this_arena_on_death_row{this};
410
411 const auto disposer_func = [this_arena_on_death_row]
412 (const Log_context_mt* log_ctx, auto&& destroy_arena_func) mutable
413 {
414 /* Careful. This might be running synchronously from destroy_on_obj_db_forgot_us() (which itself might
415 * be, or might not be, running synchronously from destroy()), or it might be running from another thread
416 * due to straggling Thread_cache-owning thread(s) needing to flush stuff first. log_ctx is valid either
417 * way; if synchronously from destroy_on_obj_db_forgot_us() then it is that guy's passed-in one; if async
418 * then it is some Thread_cache's Log_context_mt instead. (Meanwhile we reiterate: Ipc_arena's get_logger()
419 * is null. Again: log_ctx or bust from now on.)
420 *
421 * If `m_arenas.size() == 1`, then there is one disposer_func copy and one this_arena_on_death_row copy.
422 * So it'll reach ref-count 0 below, causing finally the destruction of the Ipc_arena *this.
423 *
424 * If `m_arenas.size() > 1`, then there are a couple possibilities. There are then N disposer_func()
425 * (and this_arena_on_death_row) copies out there, and they might execute serially or (at least partially)
426 * concurrently. If serially: nothing special to say. If concurrency occurs:
427 * Ipc_arena and its super-classes can handle concurrent extent hooks executing; after all
428 * an allocate() might coincide with another in another thread and a deallocate() in another, etc.
429 * And that's for the *same* jemalloca-arena (e.g., m_arena0); if they're separate (the case here)
430 * then definitely fine. / Regardless: this_arena_on_death_row is 1 of N copies, so if it's the last
431 * of the N to be handled, then ref-count 0 belos is reached + *this dies. Otherwise not. */
432
433 FLOW_LOG_SET_LOCKED_CONTEXT(log_ctx);
434 FLOW_LOG_INFO_LOCKED("Ipc_arena death-row disposer executing for one of the jemalloc-arenas; it is safe to "
435 "destroy. For internal reasons logging from resulting extent-hooks firing during this "
436 "arena-destruction has been disabled. Destroying jemalloc arena now.");
437
438 /* A bunch of SHM-pool removals (might) occur synchronously in here, triggering hooks => calling
439 * stuff in *this Ipc_arena which may well log... but we've set get_logger() to null in destroy(); so
440 * sadly that stuff will not be visible.
441 *
442 * @todo It would be better if it were visible: if those hooks could log. It would merely require us to
443 * hook up log_ctx to those log call sites. Tactically this is a bit difficult: Consider at least two parts
444 * of *this hierarchy: Ipc_arena itself (extent hooks logging) and Shm_pool_collection super-super-class
445 * (pool-ops, like closing of pool, logging). At the moment they just use Log_context super-class's stuff
446 * for regular FLOW_LOG_INFO()/etc. calls. We'd need to set something somewhere that would cause such
447 * log-call-sites to use `Log_context_mt* log_ctx` (which we've spent some serious effort in obtaining across
448 * dangerous territory) instead of the normal thing. (Having Shm_pool_collection/Ipc_arena/et-al derive
449 * from Log_context_mt does not do it: *log_ctx might be some Thread_cache's log-context; loading the get_logger()
450 * from there into a hypothetical *this Log_context_mt defeats the point: If user does set_logger(), it will
451 * be against Thread_cache/its Log_context_mt -- not whatever we copied out of it.) It is definitely doable
452 * but requires auditing all of the potentially-affected code and having it do the proper hoop-jumping to
453 * select the proper Log_context_mt (without forgetting about perf). Worth looking into: the arena-end
454 * logging could be helpful in practice. We do at least log here ourselves (but the hooks won't). */
455 destroy_arena_func();
456
457 /* If `m_arenas.size() == 1` then: The following *will* trigger the default deleter, namely,
458 * delete this_arena_on_death_row.get()
459 * If `m_arenas.size() > 1` then: The following *might* do that, but across the disposer_func() copies
460 * it'll do that exactly once.
461 *
462 * (We could omit the line, but it'd happen momentarily anyway. This way at least it's nice for
463 * debugger sessions.) */
464 this_arena_on_death_row.reset();
465 }; // const auto disposer_func =
466 using Disposer_func = decltype(disposer_func);
467
468 /* So now, for each arena_id in series, it will either actually destroy the arena right now or defer it
469 * until thread(s) get a chance to do it. Since we still have this_arena_on_death_row on the stack, `*this` won't be
470 * destroyed within the loop, even if all of the arenas are such that they got destroyed synchronously. */
471 for (const auto arena_id : m_arenas)
472 {
473 FLOW_LOG_INFO_LOCKED("Destroying arena (possibly deferred due to tcache dependency), "
474 "jemalloc-ID [" << arena_id << "].");
475 get_jemalloc_memory_manager()->destroy_arena(arena_id, log_ctx,
476 // Make a copy, as it's a destructive && arg.
477 Disposer_func{disposer_func});
478 }
479
480 assert(this_arena_on_death_row);
481 /* Now this_arena_on_death_row will go out of scope, so it's possible `delete this` happens right here.
482 * E.g.: `m_arenas.size() == 1`; Thread_cache was only ever used (allocations only ever happened) right in this
483 * thread; so the single arena just got destroyed. */
484} // Ipc_arena::destroy_on_obj_db_forgot_us()

566arena_id_t Ipc_arena::create_jemalloc_arena()
567{
568 using ipc::shm::arena_lend::detail::Shm_pool_offset_ptr_data_base;
569 using detail::stat::mallctl_read;
570 using flow::util::ostream_op_string;
571 using flow::error::Runtime_error;
572 using boost::system::system_category;
573 namespace errc = boost::system::errc;
574
575 /* Check opt.retain up-front, before creating the arena. We need it: without it jemalloc refuses to set
576 * retain_grow_limit (ENOENT) -- see below for why we must set that. Once the arena exists -- wired to our
577 * extent hooks -- there is no clean way to back out of a failure; so detect this likeliest cause of such a
578 * failure first. (opt.retain is on by default in 64-bit Linux; it can be turned off via jemalloc's run-time
579 * configuration, e.g., the MALLOC_CONF environment variable.) */
580 bool retain = false;
581 if (const int ec = mallctl_read("opt.retain", &retain); ec != 0)
582 {
583 throw Runtime_error{Error_code{ec, system_category()}, "je_mallctl() error reading opt.retain."};
584 }
585 if (!retain)
586 {
587 throw Runtime_error{errc::make_error_code(errc::not_supported),
588 "jemalloc opt.retain is disabled (perhaps via run-time configuration such as "
589 "MALLOC_CONF); SHM-jemalloc requires it."};
590 }
591
592 const arena_id_t arena_id = get_jemalloc_memory_manager()->create_arena(&m_extent_hooks_wrapper); // Throws.
593
594#ifndef FLOW_OS_LINUX
595 static_assert(false, "The logic below might be Linux-specific and has only been tested in Linux as of this "
596 "writing.");
597#endif
598 /* Set the arena's retain_grow_limit to Shm_pool_offset_ptr_data_base::max_pool_size().
599 *
600 * jemalloc grows an arena's mapped vaddr footprint via a series of extent-alloc requests (for us: each a new
601 * SHM-pool), each geometrically larger than the last; retain_grow_limit caps the size of each such growth step.
602 * (This geometric growth happens only in jemalloc's retain mode: hence the name; hence jemalloc's refusal to set
603 * it without opt.retain; hence our check above.) The default is effectively no limit.
604 *
605 * Why we set it:
606 * - Hard correctness bound. Our fancy-pointer representation cannot encode addresses within a pool larger
607 * than max_pool_size() (~2Gi bytes as of this writing). This caps only the growth steps, though: a single
608 * allocation request too large for any step makes jemalloc request an extent sized for that request
609 * regardless. create_shm_pool() (the extent-alloc hook) declines any such too-large extent, which completes
610 * the guarantee.
611 * - Defensive bound. SHM objects are subject to OS-imposed size/count limits that ordinary heap mappings are
612 * not. Without a cap, an arena can grow a pool to many GBs of vaddr even when RAM-resident usage is much
613 * smaller -- observed in the field, in fact, prompting the original choice of this value.
614 *
615 * Despite the name, this is unrelated to the `stats.*.retained` stat, which counts vaddr jemalloc keeps mapped for
616 * later reuse (for us: pools, or ranges thereof, whose removal optional_remove_shm_pool() declined, their physical
617 * pages having been released). */
618 size_t input_param = Shm_pool_offset_ptr_data_base::max_pool_size();
619 const int ec
620 = IPC_SHM_ARENA_LEND_JEMALLOC_API(mallctl)(ostream_op_string("arena.", arena_id, ".retain_grow_limit").c_str(),
621 nullptr, nullptr, &input_param, sizeof(input_param));
622 if (ec != 0)
623 {
624 /* The arena exists and is wired to our extent hooks (see up-front opt.retain check above for why that makes a
625 * clean back-out impractical). With opt.retain checked, a failure here means a bad arena ID or limit value:
626 * a bug. */
627 const Error_code sys_err_code{ec, system_category()};
628 FLOW_LOG_FATAL("Setting retain_grow_limit to [" << input_param << "] for new jemalloc-arena [" << arena_id << "] "
629 "failed despite opt.retain check: system error [" << sys_err_code << "] "
630 "[" << sys_err_code.message() << "]. Bug?");
631 assert(false && "Setting retain_grow_limit failed despite opt.retain check: bug?");
632 std::abort();
633 }
634
635 return arena_id;
636} // Ipc_arena::create_jemalloc_arena()

668void* Ipc_arena::allocate_impl(size_t size)
669{
670 assert(!m_arenas.empty() && "start() must have been called by now.");
671 const auto& arena_id = m_arena0;
672
673#if IPC_SHM_ARENA_LEND_JEMALLOC_NO_TCACHE
674 // ^-- Tcache support is stable but turning it off can simplify specific profiling/debugging.
675 if (skip_fast_path_verbose_logging())
676 {
677 return get_jemalloc_memory_manager()->allocate_no_tcache(size, arena_id);
678 }
679 // else
680
681 FLOW_LOG_DATA("Allocating size [" << size << "], arena [" << arena_id << "], no tcache.");
682 void* const buf = get_jemalloc_memory_manager()->allocate_no_tcache(size, arena_id);
683 FLOW_LOG_DATA("Allocated size [" << size << "], arena [" << arena_id << "], no tcache, "
684 "resulting in [" << buf << "].");
685 return buf;
686#else // #if !IPC_SHM_ARENA_LEND_JEMALLOC_NO_TCACHE
687 // Suggest reading Thread_cache class doc header for background on jemalloc-tcache.
688
689 /* Get (and first create if needed) cache object (Thread_cache) for this thread.
690 * Get (and first create if needed) jemalloc tcache for this thread for this arena.
691 *
692 * Reminder: jemalloc docs don't make this clear, and it doesn't appear to be widely known, but both
693 * empirical evidence and links like https://github.com/jemalloc/jemalloc/issues/460 show that, while
694 * a given tcache (and tcache ID X) can be used to jemalloc-mallocx() against 2+ arenas A and B, this leads
695 * to potential trouble: such an mallocx(B) shall absolutely sometimes yield an address in an A-owned pool.
696 * For us this is totally unacceptable. (Perhaps when allocating is always in heap, without arenas ever dying
697 * throughout, it would have been fine; not fine for us at all.)
698 *
699 * It does make sense, if one has the properly simple conception of how jemalloc-tcache
700 * works; namely that it is essentially nothing more than a thread-local intermediary store of
701 * what would normally be mallocx()-returned pointers, sitting between (1) us with our actual-mallocx() calls
702 * (get_j_m_m()->allocate() above) and actual-dallocx() calls (in this->deallocate()) and (2) jemalloc and its
703 * jemalloc-arenas (this arena_id but also all the other ones). So if I do an initial mallocx() here,
704 * it might seed this thread's TL-cache with (across a single mutex-lock) hundreds of
705 * what-would-normally-be-mallocx()es' returned pointers from arena_id, yes. But if I do another initial mallocx()
706 * here, for another arena -- arena_id2 -- it may well do the same thing; now this TL-cache of allocated-buffer
707 * pointers has pointers into arena_id and arena_id2; jemalloc's tcache does not auto-segregate by source-arena.
708 * Now a subsequent mallocx(arena_id) might see the cache has lots of juicy pointers to give out -- no need to
709 * lock any mutex to do it -- and will hand out one of the ones in the TL-cache... which may well have come
710 * from arena_id2. As stated before, we can't have that: arena_id2's Ipc_arena might go away, while arena_id's
711 * Ipc_arena is still around; the (one would expect) perfectly-alive mallocx()-result pointer from (one would
712 * expect) arena_id (which is indeed perfectly alive) now points at destroyed-arena-backed garbage. */
713 const auto tcache_id_for_arena = Thread_cache::this_thread_cache()->id(arena_id);
714
715 if (skip_fast_path_verbose_logging())
716 {
717 return get_jemalloc_memory_manager()->allocate(size, arena_id, tcache_id_for_arena);
718 }
719 // else
720
721 FLOW_LOG_DATA("Allocating size [" << size << "], arena [" << arena_id << "], "
722 "tcache [" << tcache_id_for_arena << "].");
723 void* const buf = get_jemalloc_memory_manager()->allocate(size, arena_id, tcache_id_for_arena);
724 FLOW_LOG_DATA("Allocated size [" << size << "], arena [" << arena_id << "], tcache "
725 "[" << tcache_id_for_arena << "], resulting in [" << buf << "].");
726 return buf;
727#endif // #elif !IPC_SHM_ARENA_LEND_JEMALLOC_NO_TCACHE
728} // Ipc_arena::allocate()

730void Ipc_arena::deallocate(void* address) noexcept
731{
732 assert(!m_arenas.empty() && "start() must have been called by now.");
733 const auto& arena_id = m_arena0;
734
735 /* Careful: no logging -- nor any other get_logger() access -- in this method, unlike in allocate() and others.
736 * Reason: obj-db-deferred deallocations can make this method run, via an obj-db scan in some thread,
737 * concurrently with destroy()'s set_logger(nullptr) (see notes there): destroy() can begin while such a scan
738 * is in flight, and scans stop only at the _admin::forgetting_shm_arena() barrier -- which is necessarily
739 * after the set_logger(nullptr). So a get_logger() here would race that mute. allocate() is immune: it cannot run
740 * once the handle group hits ref-count 0 (no construct()ed objects remain to allocate() things). */
741
742#if IPC_SHM_ARENA_LEND_JEMALLOC_NO_TCACHE
743 get_jemalloc_memory_manager()->deallocate_no_tcache(address);
744#else // #if !IPC_SHM_ARENA_LEND_JEMALLOC_NO_TCACHE
745 // Suggest reading Thread_cache class doc header for background on jemalloc-tcache.
746
747 /* It would seem we could, much as in allocate(), just do
748 * tcache_id_for_arena = Thread_cache::this_thread_cache()->id(arena_id);
749 * so that the Thread_cache and inside it the per-arena jemalloc-tcache are created if not currently in existence.
750 * We intentionally avoid this, treating this thread's Thread_cache read-only (our data structure, not
751 * the jemalloc-internal tcache: it is certainly potentially written-to -- if it exists, but the point is
752 * we don't make it exist, if it doesn't already). Reason:
753 *
754 * In many cases, in and of itself, it would be fine and good. Say allocate(arena_id) returned `address` in
755 * thread T1, and we are deallocate()ing in thread T2, and one has never allocate(arena_id)ed in T2 so far.
756 * Creating the tcache here would push `address` into a new tcache, and further deallocate(<thing from arena_id>)s
757 * would keep pushing onto it sans mutex-locking -- better than the normal mutex-locking no-tcache dallocx().
758 *
759 * The danger area is during, generally speaking, "things" going down -- the user would not be triggering
760 * deallocate() themselves, but our obj-DB tracking system might. 1, at thread exit
761 * detail::Thread_lcl_obj_db_admin can (much as during thread's lifetime) delete construct()ed objects that
762 * lack live user handles (a/k/a zombie objects; note these do not represent an error or anomaly, despite the
763 * name). 2, when *this Ipc_arena is going down (destroy() path), it unconditionally similarly deletes
764 * any still-live construct()ed objects. For (1), that thread's Thread_cache might have been cleaned-up
765 * already, so at least the outer deallocate() during said deletion would re-create Thread_cache. It should
766 * still be cleaned-up again quite soon by Thread_local_state_registry/Thread_local_ptr/thread_specific_ptr
767 * stubborn-cleanup machinery, but this is hard to reason about and harder to maintain over time. For (2),
768 * as of this writing it is probably safe actually: first
769 * Thread_lcl_obj_db_admin::forgetting_shm_arena() does the deletes, then triggers the relevant Thread_cache
770 * cleaning, so it should be still around when needed (during the deletes). Again, though, it is hard to
771 * reason about and maintain. We just don't want that trouble.
772 *
773 * There is a cost to this, all else being equal: Until the first arena_id allocate() in thread T,
774 * any deallocate()s (of things allocate()d in other threads: totally possible and common enough) will
775 * do the full-mutex-locking tcache-bypassing operation that returns the buffer to jemalloc's arena-set.
776 * With an active tcache, it could skip that and return to the tcache intermediary store instead. This is a
777 * real cost, but it is well mitigated.
778 * - Consider a given deallocate(). Often (not always), it is part of an outer-object's
779 * deletion by Thread_lcl_obj_db_admin (TLODBA): at least the deallocate() of the object's raw sizeof().
780 * If not (e.g., an in-SHM `vector` being move-assigned-to => deletion of overwritten buffer), the same
781 * thread T probably (not always) would've TLODBA-deleted *some* outer object at least once. So what?
782 * They're still DEallocations; doesn't help. True: but TLODBA-deletion of object X in thread T
783 * means X *must* have been construct()ed in thread T. If so, then the construct() would first-thing
784 * have allocate(sizeof())d... and that first-thing would have create the tcache after all.
785 * - Supposing the "often" and "probably" in the prev bullet are accurate, there is the possible corner case
786 * (totall allowed and not an error though) that TLODBA-deletion in thread T is actually handling object X
787 * that was created in another thread T2, not T: but this specifically only happens in the
788 * degraded-admin drain-thread TLODBAs. For that reason
789 * Thread_lcl_obj_db_admin::degraded_admin_thread_body(), in controlled fashion, pre-creates the tcache
790 * ahead of any TLODBA-deletes it would then do.
791 *
792 * @todo This does still leave a hole as noted: It is possible -- hard to say how common but "feels" not-too-common --
793 * that some thread might *just* deallocate() lots and lots, without allocate()ing once throughout. It is OK:
794 * it just won't get the tcache perf-bonus for those deallocate()s. Nevertheless it'd be nice to plug the hole.
795 * All it would take is either some heuristic... but let's aim higher in this to-do... or knowing to be
796 * read-only (like now) only in the danger-zone -- we think, as explained above, around thread exit and
797 * arena destruction. Any mechanism for determining this is TBD; it is unlikely to be trivial but feels doable.
798 * The degraded_admin_thread_body() heuristic can then also be removed. */
799
800 const auto this_thread_cache_or_null = Thread_cache::this_thread_cache_or_null();
801 const auto tcache_id_for_arena = this_thread_cache_or_null ? this_thread_cache_or_null->id_or_none(arena_id)
802 : Thread_cache::S_NO_TCACHE_ID;
803
804 if (tcache_id_for_arena != Thread_cache::S_NO_TCACHE_ID)
805 {
806 /* In this thread, tcache (for relevant arena) exists. Use it -- regardless of which thread allocated
807 * the buffer: a tcache is merely a thread-local store of pointers to this arena's regions awaiting
808 * reuse; a region carries no memory of the tcache (if any) that handed it out; so any this-arena region
809 * may be sitting in the calling thread's tcache. (See discussion at end of function for how it comes to
810 * be -- and how commonly -- that the allocating thread differs from ours.) */
811
812 get_jemalloc_memory_manager()->deallocate(address, tcache_id_for_arena);
813 return;
814 } // if (tcache_id_for_arena != Thread_cache::S_NO_TCACHE_ID)
815 // else: No tcache available in this thread (yet), for this arena.
816
817 /* Do a similar thing to the NO_TCACHE snippet above. (Slight perf loss results versus the tcache path
818 * above -- the alloc was tcache-enabled, but this thread has not requested tcache creation for the arena,
819 * presumably having never allocated in it at this layer.) */
820 get_jemalloc_memory_manager()->deallocate_no_tcache(address);
821
822 /* Discussion for context: Suppose end user's actual code *outside any SHM-aware allocators* refrains from making
823 * direct [de]allocate() calls (which isn't disallowed either incidentally). That represents not-guaranteed
824 * but definitely recommended behavior. So in that case, can deallocate() even occur in a thread other than
825 * the buffer's allocate() thread? (Not that anything above requires a particular answer; but it affects
826 * tcache-usage patterns, so it is worth understanding.)
827 *
828 * At first it probably seems like it's not, and that would be good; various reasoning becomes much simpler.
829 * It seems like it's not possible: user must do construct<T>() which triggers allocate(sizeof(T).
830 * Thread_lcl_obj_db* machinery is built on the idea that *once it is determined the T goes out of scope, in
831 * whichever thread or process*, only the original constructing thread will (opportunistically) destroy the T,
832 * calling deallocate(). As of this writing you can see that in construct() where it passes a really-destroy-T
833 * functor to Thread_lcl_obj_db_admin->constructing_obj(). Awesome!
834 *
835 * Indeed, when T is simple, without allocating/deallocating buffers on its behalf, the above holds, and that's
836 * great. It might not hold, though, if T allocates/deallocates on its behalf. We're assuming things are done
837 * in recommended fashion (assumption at top of this comment); in which case T is/contains container(s).
838 * Take 1 container, a vector<char>. In the simplest case one might construct<T>(10): so the outer [de]allocate()
839 * of sizeof(T) occurs as explained above, all good. In addition though when T{10} executes it'll
840 * ask the SHM-aware allocator (our Stateless_allocator recommended) to allocate 10. Allocator calls
841 * allocate(10), that same thread: cool. Eventually handle to the T go away, and _admin deallocate(sizeof(T))s
842 * as explained, then calls ~T(); that asks Stateless_allocator to deallocate the 10; allocator
843 * thus calls deallocate(). Awesome! All still in the proper thread.
844 *
845 * However, one could resize the vector in some other thread in-between, requiring a bigger buffer. That'll
846 * deallocate() (<-- "wrong" thread), allocate() ("wrong" thread). At that point the possibilities are endless;
847 * they could let it live until ~T(); then the 2nd allocate() is matched with original-thread deallocate()
848 * (mismatch). Or they could do more allocs/deallocs, in various threads that may or may not match each other.
849 * Bottom line: If the user does allocs/deallocs via T=vector<char>, and happens to do so in other threads than
850 * where they chose to originally construct() => mismatch. Totally possible.
851 *
852 * For completeness (?) consider also that one need not even use construct(). One can declare a
853 * `vector<char, Stateless_allocator<...>> v{100, 'A'};` on the stack, or similarly on the heap. The SHM-allocating
854 * allocator will do allocate(100) there. Granted, why do this, if this `v` cannot be lent/borrowerd (shared
855 * with different process -- the point of using SHM)? Well, by itself, true; but it's allowed and normal
856 * to have such temp guys sitting around; one can then construct() -- in some other thread -- another vector
857 * and move v's buffer into it: `construct<vector<...>>(std::move(v))`. Or maybe it'd be more realistically into
858 * a construct()ed struct with a vector<char, Stateless_allocator<...>> member. Either way: might be different
859 * thread than the construct()ing one -- in this case the buffer in question even predates the construct()!
860 *
861 * So that's how. */
862#endif // #elif !IPC_SHM_ARENA_LEND_JEMALLOC_NO_TCACHE
863} // Ipc_arena::deallocate()

944 bool* commit, arena_id_t arena_id)
945{
946 using ipc::shm::arena_lend::detail::Shm_pool_offset_ptr_data_base;
947 using flow::util::stat::fetch_add;
948 using std::string;
949 using std::shared_ptr;
950
951 assert(zero);
952 assert(commit);
953
954 if (size > Shm_pool_offset_ptr_data_base::max_pool_size())
955 {
956 /* jemalloc's retain_grow_limit (see create_jemalloc_arena()) keeps its routine growth steps within this
957 * limit; but a single allocation request too large for that makes jemalloc ask for an extent sized for it
958 * regardless. We cannot encode pointers into such a pool (offsets past the limit), so decline; jemalloc then
959 * fails the allocation (=> allocate() throws bad_alloc). */
960 FLOW_LOG_WARNING("Asked to create SHM pool of size [" << size << "], arena [" << arena_id << "], exceeding the "
961 "max pool size [" << Shm_pool_offset_ptr_data_base::max_pool_size() << "] addressable by "
962 "our fancy-pointer type. Probably a single allocation of (nearly) that size or more was "
963 "attempted. Declining; the allocation shall fail.");

968 /* Ultra-unique ID generated here. It's used as a key in various maps including borrower-side global maps.
969 *
970 * For context: at least Lend_tracker_pool uses the same pool-ID namespace (same pool name prefix for a given
971 * Ipc_arena generating these "real" SHM-pools and the Lend_tracker_pool "aux" SHM-pools that we use to track
972 * objects within the "real" ones; both use this generator function). This at least allows wholesale cleanup
973 * (via file-system) of both types of things when needed. If we had some other type of aux pool, we could
974 * have it use the same technique for the same reason(s). */
975 const auto id = Shm_pool_offset_ptr_data_base::generate_pool_id();
976 // It may also be encoded in the name, to provide uniqueness and for convenience in debugging and such.
977 const string name = generate_shm_object_name(id).str(); // @todo Should be using Shared_name ~throughout.
978
979 shared_ptr<Shm_pool> pool
980 = Owner_shm_pool_collection::create_shm_pool(id, name, size, address,
981 [&](int fd, size_t size, void* address) -> void*
982 {
983 void* const actual_address = Jemalloc_pages::map(address, size, alignment, *commit, fd);
984 if (!actual_address)
985 {
986 return nullptr; // Our caller cleans up (closes `fd`, removes SHM object); we then log a WARNING.
987 }
988
989 // Extracted (somewhat) from extent_mmap.c
990 *zero = *commit;
991
992 FLOW_LOG_TRACE("Mapped SHM pool at address [" << actual_address << "].");
993
994 return actual_address;
995 });
996
997 void* pool_address;
998 if (pool)
999 {
1000 pool_address = pool->get_address();
1001 FLOW_LOG_TRACE("Created SHM pool at address [" << pool_address << "], name [" << name << "], "
1002 "size [" << size << "], arena [" << arena_id << "].");
1003
1004 { // Stats. (See also to-do in the `else` branch.)
1005 fetch_add(&m_pool_stats.m_owner_pool.m_pool_create_count, 1);
1006 fetch_add(&m_pool_stats.m_owner_pool.m_pool_create_sz, size);
1007 }
1008 }
1009 else
1010 {
1011 // Failed to create pool.
1012 pool_address = nullptr;
1013 FLOW_LOG_WARNING("Failed to create SHM pool of size [" << size << "].");
1014
1015 /* @todo Stat-member doc header says pool-create request is always satisfied. This WARNING/code-path looks to me
1016 * (ygoldfel) like it indicates pretty much a catastrophic/unexpected situation that is hardly recoverable.
1017 * We are somewhere in the middle of a jemalloc alloc itself in the middle of a user SHM-alloc, too, so
1018 * it's not like reporting the problem (whatever it even might be) to the user is straightforward -- how *would*
1019 * we even do it, and what % of users would even try to handle this tiny possibility every time?
1020 * The point: Probably we should FLOW_LOG_FATAL() and std::abort() here and in places like it.
1021 * There's probably a bigger-scope to-do to that effect elsewhere (as SHM-jemalloc is littered with attempts
1022 * at civilized handling of errors that doesn't really reach the user in an intended way and is therefore
1023 * arguably more trouble than a crash-out); I nevertheless wanted to emphasize this particular one.
1024 *
1025 * Historical note: Code originally written by another developer (echan), and while we have refactored much of
1026 * *this class since then, this extent-hook area is quite deep/low-level, and generally works well, so beyond
1027 * cosmetic tweaks we have not touched it in similar ways. This to-do is a way in which there is still some
1028 * refactoring (at least arguably) still left. */
1029 }
1030
1031 const auto logger = get_logger();
1032 if (logger && logger->should_log(flow::log::Sev::S_TRACE, get_log_component()))
1033 {
1034 print_shm_pool_map();
1035 }
1036
1037 return pool_address;
1038} // Ipc_arena::create_shm_pool()

1102 shm_pool->get_fd(), offset, length, true);

1245
1246 shared_ptr<Shm_pool> pool_a = lookup_shm_pool(address_a);
1247 if (!pool_a)

1572std::vector<Ipc_arena::Memory_manager_stats> Ipc_arena::memory_manager_stats()
1573{
1574 using detail::stat::config_stats_enabled;
1575 using flow::util::stat::stats_since_reset_state;
1576 using std::vector;
1577
1578 /* m_arenas is a std::set => iteration is ascending by native (jemalloc) arena-ID; hence the result is sorted
1579 * that way, as promised by our contract. The arena-identifying IDs are ours to know regardless of jemalloc's
1580 * build flags; set them on every row. */
1581 vector<Memory_manager_stats> result_id_sorted(m_arenas.size());
1582 const auto uniq_id = uniq_arena_id();
1583 {
1584 size_t idx = 0;
1585 for (const auto arena_id : m_arenas)
1586 {
1587 auto& row = result_id_sorted[idx++];
1588 row.m_uniq_arena_id = uniq_id;
1589 row.m_native_arena_id = uint64_t(arena_id);
1590 }
1591 }
1592
1593 /* The jemalloc-sourced gauges require a stats-enabled jemalloc build. If absent: still emit a row per arena
1594 * (with IDs set), but leave the real stat-members at defaults (zeroes probably) -- as advertised. */
1595 if (!config_stats_enabled())
1596 {
1597 return result_id_sorted;
1598 }
1599 // else: populate the raw stat-members, then fold them into the since-reset view (including HI_WMARKs).
1600
1601 // Lock throughout, including the raw reads; see detail::jemalloc_stats_mutex() and m_mem_mgr_base_stats doc headers.
1602 Lock lock{detail::jemalloc_stats_mutex()};
1603
1604 memory_manager_snapshot_read(&result_id_sorted);
1605
1606 /* Formally speaking we do here what flow::util::stat::stats_since_reset_state() doc header prescribes:
1607 * read the raw values from data-source (jemalloc-stats), then do the following against -- and potentially
1608 * updating -- m_mem_mgr_base_stats[arena_id]. As a reminder though: ACCUMULATORs are -=ed against the base values;
1609 * GAUGEs are left alone; HI_WMARKs are chosen as the higher of the new GAUGE value and the one in
1610 * m_mem_mgr_base_stats; and lastly if the former then the HI_WMARK in base struct is updated to the new max too
1611 * (for next time we do this if any). */
1612 size_t idx = 0;
1613 for (const auto arena_id : m_arenas)
1614 {
1615 stats_since_reset_state(&result_id_sorted[idx++], &(m_mem_mgr_base_stats[arena_id]));
1616 }
1617
1618 return result_id_sorted;
1619} // Ipc_arena::memory_manager_stats()

1621void Ipc_arena::memory_manager_stats_reset()
1622{
1623 using detail::stat::config_stats_enabled;
1624 using flow::util::stat::stats_mark_reset_state;
1625 using std::vector;
1626
1627 /* Formally speaking we do here what flow::util::stat::stats_since_reset_state() doc header prescribes.
1628 * Read that guy please for details of how it all works. Basically, though, we are to grab the raw
1629 * values for the data-source, which results in the following items used in subsequent stat-consumptions
1630 * (memory_manager_stats()): ACCUMULATOR saved for a delta computation; HI_WMARK saved to current GAUGEd
1631 * value, since by definition the max since right-now = value right-now. That last part is achieved
1632 * via stats_mark_reset_state().
1633 *
1634 * Only meaningful with a stats-enabled jemalloc; otherwise there is nothing to read (bases stay zero). */
1635 if (!config_stats_enabled())
1636 {
1637 return;
1638 }
1639 // else:
1640
1641 vector<Memory_manager_stats> rows(m_arenas.size());
1642
1643 // Lock throughout, including the raw reads; see detail::jemalloc_stats_mutex() and m_mem_mgr_base_stats doc headers.
1644 Lock lock{detail::jemalloc_stats_mutex()};
1645
1646 memory_manager_snapshot_read(&rows);
1647
1648 /* Install current raw reading as the new baseline, then mark (sets each HI_WMARK = its gauge => running
1649 * max restarts at "now"). */
1650 size_t idx = 0;
1651 for (const auto arena_id : m_arenas)
1652 {
1653 auto& base = m_mem_mgr_base_stats[arena_id];
1654 auto& row = rows[idx++];
1655 base.m_vaddr = std::move(row.m_vaddr);
1656 base.m_alloc = std::move(row.m_alloc);
1657 stats_mark_reset_state(&base);
1658 }
1659} // Ipc_arena::memory_manager_stats_reset()

1661void Ipc_arena::memory_manager_snapshot_read(std::vector<Memory_manager_stats>* rows) const
1662{
1663 using detail::stat::epoch_refresh;
1664 using detail::stat::epoch;
1665
1666 assert(rows && (rows->size() == m_arenas.size()));
1667
1668 // See our doc header for the reasoning behind all this. (Pre-condition: detail::jemalloc_stats_mutex() locked.)
1669 auto snapshot_epoch = epoch_refresh();
1670 while (true)
1671 {
1672 size_t idx = 0;
1673 for (const auto arena_id : m_arenas)
1674 {
1675 auto& row = (*rows)[idx++];
1676 memory_manager_vaddr_stats_read(arena_id, &row.m_vaddr);
1677 memory_manager_alloc_stats_read(arena_id, &row.m_alloc);
1678 }
1679
1680 const auto epoch_now = epoch();
1681 if (epoch_now == snapshot_epoch)
1682 {
1683 return; // Nobody refreshed while we were reading: all from 1 snapshot.
1684 }
1685 // else
1686
1687 FLOW_LOG_INFO("Ipc_arena [" << this << "]: jemalloc stats were refreshed elsewhere (epoch "
1688 "[" << snapshot_epoch << "] => [" << epoch_now << "]) while we were reading them; reading them "
1689 "again. This is normal but should be infrequent.");
1690 snapshot_epoch = epoch_now;
1691 } // while (true)
1692} // Ipc_arena::memory_manager_snapshot_read()

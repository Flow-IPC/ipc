31std::ostream& operator<<(std::ostream& os, const Serializer_info_dump& val)
32{
33 using flow::util::ostream_op_string;
34 using flow::util::stat::print;
35 using util::String_view;
36 using std::string;
37
38 /* Core's rcv-side is inherently unused (see Core_serializer_stats doc header): skip it when printing, for
39 * signal/noise's sake. print() emits the `m_rcv` stats last, each named "rcv.<...>"; so: print to a string;
40 * truncate at the first " rcv." occurrence. It's a bit haxory but correct; and the assert()s guard against the
41 * underlying assumptions changing. */
42
43 // If these ever trip, core's rcv-side is used after all: stop hiding it here (and update its doc headers).
44 assert((val.m_core.m_rcv.m_alloc_lifetime_sz.load() == 0)
45 && (val.m_core.m_rcv.m_msgs_outstanding_hi_wmark.load() == 0)
46 && "Core_serializer_stats rcv-side has real data?! Un-hide it here.");
47 string core_str = ostream_op_string(print(val.m_core));
48 const auto rcv_pos = core_str.find(" rcv.");
49 assert((rcv_pos != string::npos) && "print(Serializer_stats) rcv-section marker not found; format changed?");
50 core_str.erase(rcv_pos);
51
52 String_view ln; // Separator when next thing has no conceptual indent-level.
53 String_view ln_ln; // Separator when next thing has 1 conceptual indent-level.
54 if (val.m_fmt.m_multiline)
55 {
56 os << "- ";
57 ln = "\n- ";
58 ln_ln = "\n - ";
59 }
60 else
61 {
62 ln_ln = (ln = " | ");
63 }
64
65 return os << "heap-backed: [" << print(val.m_heap) << ']' << ln
66 << "shm-backed->" << ln_ln
67 << "heap-backed-envelope: [" << print(val.m_outer) << ']' << ln_ln
68 << "shm-backed-core: [" << core_str << ']';
69 // (Intentional: no newline at the end.)
70} // operator<<(ostream&, Serializer_info_dump)

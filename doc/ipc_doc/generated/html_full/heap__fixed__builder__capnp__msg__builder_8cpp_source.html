35Heap_fixed_builder_capnp_message_builder::Heap_fixed_builder_capnp_message_builder
36 (size_t seg_and_frame_sz_cap, size_t frame_prefix_sz1, size_t frame_prefix_sz_cont,
37 bool grow_seg_sz_else_constant, size_t segment_sz_init_pre_growth,
38 stat::Serializer_stats::Snd* stats) :
39 /* This is a hard cap for a thing that must be a word-multiple; so we want it to be a word-multiple;
40 * but also this is an attempt to fit every segment in this many bytes at most *including* the frame prefix.
41 * Point is, round down: do not use round_to_multiple() (which rounds up). */
42 m_seg_and_frame_sz_cap((seg_and_frame_sz_cap / sizeof(::capnp::word)) * sizeof(::capnp::word)),
43 m_frame_prefix_sz1(frame_prefix_sz1),
44 m_frame_prefix_sz_cont(frame_prefix_sz_cont),
45 m_grow_seg_sz_else_constant(grow_seg_sz_else_constant),
46 // Borrow MallocMessageBuilder's heuristic when grow_seg_sz_else_constant (and don't forget to apply the cap).
47 m_segment_sz(grow_seg_sz_else_constant
48 ? std::min(flow::util::round_to_multiple(segment_sz_init_pre_growth, sizeof(::capnp::word)),
49 m_seg_and_frame_sz_cap - m_frame_prefix_sz1)
50 : (m_seg_and_frame_sz_cap - m_frame_prefix_sz1)),
51 m_stats(stats),
52 m_alloc_sz(0),
53 m_saw_big_leaf(false)
54{
55 // By contract we can't let the prefix frame change the start of the area given to capnp to be non-aligned.
56 assert(((m_frame_prefix_sz1 % sizeof(::capnp::word)) == 0)
57 && "N=frame_prefix_sz1 passed to ctor, by contract, must be such that (aligned_ptr + N) is also aligned.");
58 assert(((m_frame_prefix_sz_cont % sizeof(::capnp::word)) == 0)
59 && "N=frame_prefix_sz_cont passed to ctor, by contract, must be such that (aligned_ptr + N) is also aligned.");
60
61 /* Also ensure the cap leaves at least 1 word of segment space past each frame prefix; otherwise the
62 * cap-minus-prefix arithmetic here and in allocateSegment() would underflow (and note the assert just below
63 * would be satisfied by the same underflowed values, hence would not save us). */
64 assert((m_seg_and_frame_sz_cap >= (m_frame_prefix_sz1 + sizeof(::capnp::word)))
65 && "seg_and_frame_sz_cap must leave at least 1 word of segment space past the seg-1 frame prefix.");
66 assert((m_seg_and_frame_sz_cap >= (m_frame_prefix_sz_cont + sizeof(::capnp::word)))
67 && "seg_and_frame_sz_cap must leave at least 1 word of segment space past the continuation frame prefix.");
68
69 /* allocateSegment() relies on the pre-condition that m_segment_sz is capped (and perpetuates this); we (or the user,
70 * if m_grow_seg_sz_else_constant) should have ensured this pre-condition initially. */
71 assert((m_segment_sz <= (m_seg_and_frame_sz_cap - m_frame_prefix_sz1))
72 && "We should have assigned this to the cap or capped it at the cap, above.");
73}

75Heap_fixed_builder_capnp_message_builder::~Heap_fixed_builder_capnp_message_builder()
76{
77 using flow::util::stat::fetch_sub;
78 using flow::util::stat::fetch_add;
79
80 /* Stats: end-of-life sampling. No-op if no stats configured, or if `*this` was never used to
81 * actually allocate (e.g., a Msg_out was constructed but never serialized). */
82 if ((!m_stats) || m_serialization_segments_plus_frame_space.empty())
83 {
84 return;
85 }
86 // else
87
88#ifndef NDEBUG
89 const auto prev_alloc_outstanding_sz =
90#endif
91 fetch_sub(&m_stats->m_alloc_outstanding_sz, m_alloc_sz);
92 assert((prev_alloc_outstanding_sz >= m_alloc_sz)
93 && "GAUGE underflow: dtor's per-msg subtraction exceeds the cumulative outstanding.");
94
95#ifndef NDEBUG
96 const auto prev_msgs_outstanding =
97#endif
98 fetch_sub(&m_stats->m_msgs_outstanding, 1);
99 assert((prev_msgs_outstanding >= 1)
100 && "GAUGE underflow: dtor decrement on already-zero msgs-outstanding.");
101
102 /* Allocated-bytes; we've summed this opportunistically. As promised we sample this at the end, once it can
103 * no longer grow through further capnp-mutations and therefore allocateSegment()s (potentially). */
104 m_stats->m_histo_msg_alloc_sz.record_value(m_alloc_sz);
105
106 /* Used-bytes: sum of bytes capnp actually ended up using by the end. Note: if
107 * emit_segment_blobs() ran, the Basic_blobs were also resize()d to match these sizes; if it did
108 * not run (e.g., Msg_out built but never sent), getSegmentsForOutput() still reflects capnp's last
109 * write -- so this is the right source either way. Using a Basic_blob::size() here would be a (subtle, often
110 * hard-to-notice) bug. */
111 uint64_t used_bytes = 0;
112 for (const auto& capnp_seg : getSegmentsForOutput())
113 {
114 used_bytes += capnp_seg.asBytes().size();
115 }
116 m_stats->m_histo_msg_used_sz.record_value(used_bytes);
117
118 m_stats->m_histo_segs_per_msg.record_value(m_serialization_segments_plus_frame_space.size());
119} // Heap_fixed_builder_capnp_message_builder::~Heap_fixed_builder_capnp_message_builder()

121void Heap_fixed_builder_capnp_message_builder::emit_segment_blobs(Segment_ptrs* target_blob_ptrs_ptr)
122{
123 // We'll need to adjust it a bit below.
124 auto& blobs = m_serialization_segments_plus_frame_space;
125 auto& target_blob_ptrs = *target_blob_ptrs_ptr;
126
127 assert((!blobs.empty()) && "Should not be possible for serialization to be empty with our use cases. Investigate.");
128 target_blob_ptrs.reserve(target_blob_ptrs.size() + blobs.size()); // Remember: we are appending; not replacing.
129
130 const auto capnp_segs = getSegmentsForOutput();
131 assert((capnp_segs.size() == n_segments())
132 && "Somehow our MessageBuilder created fewer or more segments than allocateSegment() was called?!");
133
134 for (size_t idx = 0; idx != capnp_segs.size(); ++idx)
135 {
136 // Pointer-to-word + size in words => pointer-to-byte + size in bytes.
137 const auto capnp_seg = capnp_segs[idx].asBytes();
138 const auto seg_sz = capnp_seg.size();
139
140 auto& blob = blobs[idx];
141
142 assert((capnp_seg.begin() == blob.begin())
143 && "Somehow capnp-returned segments are out of order to allocateSegment() calls; or something....");
144 assert((seg_sz != 0)
145 && "capnp shouldn't be generating zero-sized segments.");
146
147 blob.resize(seg_sz); // Just pull up end() to immediately follow the serialization.
148
149 /* @todo Maybe output some stuff, like the segment dump, here, like
150 * shm::capnp_message_builder::lend() does? Though, in practice, this is always the
151 * top serialization and hence is printed out in pretty good detail as the actual blob is transmitted over
152 * ipc::transport core layer. So maybe skip it to avoid unneeded verbosity. */
153
154 /* Lastly output a *pointer*, as promised, to the Basic_blob we hold. See our doc header re. what they're formally
155 * allowed to do to the pointee. Basically though: all bets are off upon the next capnp-mutation; until
156 * then any read-only access is fine; as for writing they can only write into the header area
157 * (from `.begin() - .start()` and up-to `.begin()`) and can modify nothing else (no other memory area nor
158 * the Basic_blob itself). Though, as for the latter, we basically just don't want .begin() to change. */
159 target_blob_ptrs.push_back(&blob);
160 } // for (idx in [0, capnp_segs.size()))
161} // Heap_fixed_builder_capnp_message_builder::emit_segment_blobs()

163kj::ArrayPtr<capnp::word>
164 Heap_fixed_builder_capnp_message_builder::allocateSegment(unsigned int min_sz_words) // Virtual.
165{
166 using Word = capnp::word;
167 using Capnp_word_buf = kj::ArrayPtr<Word>;
168 using flow::util::stat::fetch_add;
169 using flow::util::stat::update_hi_wmark;
170 constexpr size_t WORD_SZ = sizeof(Word);
171 const size_t min_sz = size_t(min_sz_words) * WORD_SZ; // Don't forget: in their API min_sz is in `word`s.
172
173 /* It'd be nice not to realloc m_serialization_segments_plus_frame_space's internal buffer, as that involves
174 * moving `Blob`s around: constant-time/cheap though it is, if we can avoid it, good.
175 * This value is pretty decent; 1 is really most typical, while 4 shallow `Blob`s is not a ton of RAM. */
176 constexpr size_t N_SEGS_GUESS = 4;
177 m_serialization_segments_plus_frame_space.reserve(N_SEGS_GUESS); // No-op after 1st time.
178
179 /* Background from capnp: They're saying they need the allocated space for serialization to store at least min_sz:
180 * probably they're going to store some object that needs at least this much space. So typically it's some
181 * scalar leaf thing, like 4 bytes or whatever; but it could be larger -- or even huge (e.g., a Data or List
182 * of huge size, because the user mutated it so via a ::Builder). Oh, and it has to be zeroed, as by calloc().
183 *
184 * So all we *have* to allocate is min_sz exactly in that sense. But the idea is to try to allocate more, so that
185 * capnp can efficiently shove more objects in there too without calling allocateSegment() for each one.
186 * Namely:
187 * - If !m_grow_seg_sz_else_constant:
188 * As a fixed-heap builder we promised to make that the fixed size m_segment_sz.
189 * - Else:
190 * We're supposed to grow exponentially each time, so we keep track of the next size in m_segment_sz, same
191 * as capnp::MallocMessageBuilder internally does (check its source code). Plus, we apply the cap
192 * m_seg_and_frame_sz_cap.
193 * Of course, if min_sz exceeds that, then we have no choice but to allocate the larger amount min_sz. */
194
195 const bool big_leaf = min_sz > m_segment_sz;
196 const size_t seg_sz
197 = big_leaf ? min_sz
198 /* m_segment_sz is maintained to be a word-multiple.
199 * Its init value is pre-capped; and subsequently if we grow it, we cap it then (below). */
200 : m_segment_sz;
201
202 // Now just follow the logic we expounded on in the various doc headers. Keeping explanations light here.
203
204 m_serialization_segments_plus_frame_space.emplace_back();
205 auto& blob = m_serialization_segments_plus_frame_space.back();
206
207 const bool first_alloc = m_serialization_segments_plus_frame_space.size() == 1;
208 const size_t frame_prefix_sz = first_alloc ? m_frame_prefix_sz1 : m_frame_prefix_sz_cont;
209
210 blob.reserve(frame_prefix_sz + seg_sz, flow::util::CLEAR_ON_ALLOC);
211 /* Attn! capnp requires: it must be zeroed. And Blob::reserve(N, CLEAR_ON_ALLOC) we used *does* zero it.
212 * So do not memset() it. This is, at worst, syntactic sugar; at best it can bring significant perf improvements
213 * (e.g., calloc(64Ki) we've seen be ~20% faster than malloc() + memset()). */
214
215 blob.resize(seg_sz, frame_prefix_sz); // Set: size(), start(), respectively.
216 assert(blob.size() != 0);
217
218 /* Since we are supposed to grow exponentially, increase this for next time (if any), if so configured (and cap not
219 * yet reached). So then next time (if any) m_segment_sz can be used as-is (see above), no cap check. */
220 const bool prev_grow_seg_sz_else_constant = m_grow_seg_sz_else_constant;
221 if (prev_grow_seg_sz_else_constant)
222 {
223 m_segment_sz += seg_sz; // Both values are word-multiples.
224
225 // Compute the *next* cap (hence, always use m_frame_prefix_sz_cont); then apply it for next time (if any).
226
227 const auto hard_sz_cap = m_seg_and_frame_sz_cap - m_frame_prefix_sz_cont; // Values are word-multiples.
228 if (m_segment_sz >= hard_sz_cap)
229 {
230 m_segment_sz = hard_sz_cap; // (No-op if they're equal... but the next line applies too in that case.)
231 m_grow_seg_sz_else_constant = false; // Reached the cap; so next time, if any, don't worry about it.
232 }
233 }
234 // else { Configured to not grow from the start; or reached the cap in a preceding iteration. }
235
236 /* @todo MallocMessageBuilder does some bounding according to some absolute maximum. Probably we should do the same.
237 * Get back to this and follow capnp-interface reqs and/or follow what their internal logic does.
238 * Just, m_seg_and_frame_sz_cap is already a cap... but what if it is itself too huge? It's a corner case for sure.
239 * If we do handle it, it's probably easiest to just cap m_seg_and_frame_sz_cap when first setting it in ctor. */
240
241 // Stats.
242 if (m_stats)
243 {
244 if (first_alloc)
245 {
246 // First alloc for this message: update message counter, outstanding-gauge, high-water mark thereof.
247 fetch_add(&m_stats->m_msgs, 1);
248 update_hi_wmark(&m_stats->m_msgs_outstanding_hi_wmark,
249 fetch_add(&m_stats->m_msgs_outstanding, 1) + 1);
250 }
251
252 /* Capacity-lock transition: at-most-once per *this -- so this is also the count of `*this`es
253 * that hit the cap. */
254 if (prev_grow_seg_sz_else_constant != m_grow_seg_sz_else_constant)
255 {
256 fetch_add(&m_stats->m_seg_grow_cap_lock_count, 1);
257 }
258
259 // Big-leaf event: capnp asked for more than our preferred (based on capped-growth algorithm) next-segment size.
260 if (big_leaf)
261 {
262 fetch_add(&m_stats->m_big_leaf_alloc_count, 1);
263 m_stats->m_histo_big_leaf_sz.record_value(min_sz);
264 if (!m_saw_big_leaf)
265 {
266 m_saw_big_leaf = true;
267 fetch_add(&m_stats->m_msgs_with_big_leaf, 1);
268 }
269 }
270
271 /* Lifetime + outstanding alloc-byte counters (every call). Excludes frame prefix; that goes into
272 * m_frame_lifetime_sz separately. Total physical heap alloc = m_alloc_lifetime_sz + m_frame_lifetime_sz. */
273 fetch_add(&m_stats->m_alloc_lifetime_sz, seg_sz);
274 update_hi_wmark(&m_stats->m_alloc_outstanding_sz_hi_wmark,
275 fetch_add(&m_stats->m_alloc_outstanding_sz, seg_sz) + seg_sz);
276 m_alloc_sz += seg_sz; // For dtor.
277
278 // Frame-prefix bytes as mentioned above.
279 fetch_add(&m_stats->m_frame_lifetime_sz, frame_prefix_sz);
280 } // if (m_stats)
281
282 return Capnp_word_buf{reinterpret_cast<Word*>(blob.begin()),
283 reinterpret_cast<Word*>(blob.end())};
284} // Heap_fixed_builder_capnp_message_builder::allocateSegment()

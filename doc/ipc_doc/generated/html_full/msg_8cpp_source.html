39Capped_sz_capnp_message_builder::Capped_sz_capnp_message_builder(const util::Blob_mutable& seg, bool zero_it_please) :
40 m_seg(seg.data(),
41 (seg.size() / sizeof(::capnp::word)) * sizeof(::capnp::word)), // Round down to nearest word.
42 m_returned_seg(false)
43{
44 if (zero_it_please)
45 {
46 std::memset(m_seg.data(), 0, m_seg.size());

53kj::ArrayPtr<::capnp::word> Capped_sz_capnp_message_builder::allocateSegment([[maybe_unused]] unsigned int min_sz_words)
54 // Virtual.
55{
56 using flow::error::Runtime_error;
57 using Word = capnp::word;
58 using Capnp_word_buf = kj::ArrayPtr<Word>;
59
60 if (m_returned_seg)
61 {
62 // Segment 1 was not enough to store the needed data; so seg.size() (size cap) was insufficient.
63 throw Runtime_error{error::Code::S_INVALID_ARGUMENT,
64 "Capped_sz_capnp_message_builder::allocateSegment()[bad size cap]"};
65 }
66 m_returned_seg = true;
67
68 assert(((min_sz_words * sizeof(Word)) <= m_seg.size())
69 && "Should not be possible: First allocateSegment() always takes `1`; second would have thrown above.");
70
71 return Capnp_word_buf{reinterpret_cast<Word*>(m_seg.data()),
72 reinterpret_cast<Word*>(static_cast<uint8_t*>(m_seg.data()) + m_seg.size())};
73}

85arena_id_t Memory_manager::create_arena(extent_hooks_t* extent_hooks) const
86{
87 using flow::util::Lock_guard;
88 using flow::util::Mutex_non_recursive;
89 using flow::error::Runtime_error;
90 using boost::system::system_category;
91
92 arena_id_t arena_id;
93 size_t output_size = sizeof(arena_id);
94 extent_hooks_t** input_param;
95 size_t input_size;
96 if (extent_hooks)
97 {
98 input_param = &extent_hooks;
99 input_size = sizeof(extent_hooks);
100 }
101 else
102 {
103 input_param = nullptr;
104 input_size = 0;
105 }
106
107 int ec;
108 {
109 /* `arenas.create` mutates the jemalloc arena set, so serialize it against any concurrent
110 * whole-arena-set stats dump (and against arena-create/destroy). See detail::jemalloc_arena_list_mutex() doc
111 * header for the whole story. Do note that if some other code does arenas.create et al directly, then this
112 * won't beat it; but if they use our nice API here then all good. */
113 Lock_guard<Mutex_non_recursive> lock{detail::jemalloc_arena_list_mutex()};
114 ec = IPC_SHM_ARENA_LEND_JEMALLOC_API(mallctl)("arenas.create", &arena_id, &output_size, input_param, input_size);
115 }
116 if (ec != 0)
117 {
118 throw Runtime_error{Error_code{ec, system_category()}, "je_mallctl() error on creating arena"};
119 }
120
121 return arena_id;
122} // Memory_manager::create_arena()

133void Memory_manager::stats_dump_to_ostream(std::ostream& os, Stat_format format, util::String_view extra_opts) const
134{
135 using flow::util::Lock_guard;
136 using flow::util::Mutex_non_recursive;
137 using std::string;
138
139 /* jemalloc's malloc_stats_print() drives output via a write-callback invoked repeatedly with NUL-terminated
140 * chunks; route those straight to `os`. opts: null => human-readable text; "J" => JSON; plus any caller
141 * section flags appended via extra_opts. */
142 string opts{extra_opts};
143 if (format == Stat_format::S_JSON)
144 {
145 opts += 'J';
146 }
147
148 {
149 /* malloc_stats_print() refreshes jemalloc's stats and then reads many values; so firstly serialize it against
150 * Flow-IPC's other stats refresh-then-read sequences, lest they tear each other. See
151 * detail::jemalloc_stats_mutex() doc header.
152 *
153 * Secondly: malloc_stats_print() walks the jemalloc arena set unguarded, so serialize it against any concurrent
154 * arena-create and arena-destroy (which mutate that set). See detail::jemalloc_arena_list_mutex() doc header
155 * for the whole story. Do note that if some other code does arenas.create et al directly, then this
156 * won't beat it; but if they use our nice APIs then all good. (Lock order: stats mutex, then this leaf.) */
157 Lock_guard<Mutex_non_recursive> stats_lock{detail::jemalloc_stats_mutex()};
158 Lock_guard<Mutex_non_recursive> lock{detail::jemalloc_arena_list_mutex()};
159 IPC_SHM_ARENA_LEND_JEMALLOC_API(malloc_stats_print)
160 ([](void* cbopaque, const char* chunk)
161 { *(static_cast<std::ostream*>(cbopaque)) << chunk; },
162 // (^-- Capture-less lambda => function pointer conversion. Use their cbopaque dealio for access to `os`.)
163 &os, opts.empty() ? nullptr : opts.c_str());
164 }
165
166 if (format == Stat_format::S_JSON)
167 {
168 // The JSON is very compact (no white-space... and no newline); but our contract promises a trailing newline.
169 os << '\n';
170 }
171}

181void Memory_manager::log_stats_dump(flow::log::Logger* logger_ptr, Stat_format format,
182 flow::log::Sev sev, util::String_view extra_opts) const
183{
184 FLOW_LOG_SET_CONTEXT(logger_ptr, Log_component::S_SHM);
185 /* Single (multi-line) message; the dump is built (and run) only if `sev` passes the filter.
186 * With or without J[SON] option it ends in newline, so let's not add more in that spot. */
187 FLOW_LOG_WITH_CHECKING(sev,
188 "jemalloc stats dump (knobs: "
189 "[as-" << ((format == Stat_format::S_JSON) ? "JSON" : "text") << ", extra-opts"
190 "[" << extra_opts << "]]): "
191 "STATS_DUMP[[[\n" << stats_dump_to_string(format, extra_opts) << "]]]STATS_DUMP.");
192
193 /* @todo Not a big deal, at least not yet, but it'd be nice to provide `ostream << Memory_manager::Stat_format`
194 * as well as istream>> (using flow::util::istream_to_enum()). Could then just output `format` above and
195 * elsewhere; and it could be used in flow.cfg (and/or naked program_options and similar) options/settings
196 * directly. Its being an inner-class type would complicate putting those things in _fwd.hpp, so perhaps
197 * it would also become jemalloc::Memory_manager_stat_format... though that would be a breaking change; so
198 * maybe just forget about _fwd.hpp (an exception).
199 *
200 * Its use currently is so limited, that it seemed like overkill, and that may well remain the case. */
201}

39/**
40 * Stats specific to Channel::sync_request() -- the blocking synchronous request-response API
41 * available only via struc::Channel (not struc::sync_io::Channel). Composed into Channel_stats
42 * (as Channel_stats::m_sync_req) but is also a standalone Stat_set in its own right.
43 *
44 * @internal
45 * It is updated, but more saliently reset via `flow::util::stat::stats_reset()`, on its own by
46 * struc::Channel internals. Then when the user accesses struc::Channel::stats(), the two parts are
47 * put together into struc::stat::Channel_stats.
48 */
49struct Channel_sync_req_stats
50{
51 // Data.
52
53 /**
54 * Total valid Channel::sync_request() invocations. "Valid" means this discounts those calls that return
55 * null without emitting an error (called on on already-hosed channel/similar).
56 *
57 * @note It does include the situation where `sync_request()` immediately fails, thus hosing the `Channel`
58 * and emitting error. This is a small caveat unlikely to be of much import, as such a thing can
59 * happen at most once.
60 */
61 uint64_t m_count = 0;
62
63 /**
64 * Channel::sync_request() real-time latency histogram: time period from initiating non-blocking send
65 * to return (whether success, timeout, or error), in microseconds. If it counts in #m_count, it is recorded
66 * here, and vice versa. A late-arriving response (#m_late_responses) is recorded as ending when the timeout
67 * occurs (not when it arrives past the timeout).
68 *
69 * Default buckets: same config as sync_io::stat::Channel_stats::Rcv::m_histo_one_off_request_rtt_usec
70 * (see its doc header for rationale) -- after all a `sync_request()` latency *is* a one-off-request RTT:
71 * sync_request() internally layers a blocking wait atop a one-off `async_request()`.
72 *
73 * @note User can override bucket config via
74 * Channel::stats_configure_sync_req_latency_histogram() before Channel::start().
75 */
76 flow::util::stat::Histogram_counter m_histo_latency_usec
77 {sync_io::stat::Channel_stats::Rcv::S_HISTO_RTT_BUCKET_VAL0S_USEC,
78 sync_io::stat::Channel_stats::Rcv::S_HISTO_RTT_BUCKET_N_SZ_USEC};
79
80 /**
81 * Channel::sync_request() invocations (counted in #m_count) that returned
82 * transport::error::Code::S_TIMEOUT.
83 */
84 uint64_t m_timeouts = 0;
85
86 /**
87 * Count of the event wherein response arrived after Channel::sync_request() had already timed out; the
88 * response was dropped.
89 *
90 * High #m_late_responses relative to #m_timeouts indicates that the opposing process *is*
91 * responding but too slowly -- a different diagnosis than "never responded."
92 *
93 * @note A thing that increments #m_late_responses also increments #m_timeouts (but not necessarily vice versa).
94 */
95 uint64_t m_late_responses = 0;
96}; // struct Channel_sync_req_stats

98/**
99 * Cumulative stats from a struc::Channel, composing the underlying sync_io::stat::Channel_stats
100 * with `struc::Channel`-specific stats including vis-a-vis Channel::sync_request(). #m_core contains
101 * the former; #m_sync_req contains the latter (a Channel_sync_req_stats peer).
102 *
103 * A struc::Channel user obtains this via Channel::stats(); a sync_io::Channel user obtains
104 * sync_io::stat::Channel_stats via that class's own sync_io::Channel::stats() instead.
105 *
106 * @see doc header for struc::Channel::stats().
107 * @see doc header for #m_core; there are certain subtleties that may be of interest.
108 */
109struct Channel_stats
110{
111 // Data.
112
113 /**
114 * Stats from the underlying sync_io::Channel.
115 *
116 * @note While the vast majority of Channel methods are mirrored in sync_io::Channel, and vice versa,
117 * one exception is that Channel::async_request() will invoke sync_io::Channel::async_request() --
118 * but so will Channel::sync_request(). Therefore there may be counts/gauges in sync_io::stat::Channel_stats
119 * (with respect to requests generally) that one might not naively expect to be as high as observed,
120 * here in #m_core.
121 */
122 sync_io::stat::Channel_stats m_core;
123
124 /// Stats specific to Channel::sync_request().
125 Channel_sync_req_stats m_sync_req;
126}; // struct Channel_stats

141template<typename Visitor>
142void declare_stats(const flow::util::stat::Stat_name& name_prefix,
143 const Channel_stats* src_stats, Channel_stats* target_stats,
144 Visitor&& visitor)
145{
146 /* Core stats -- delegate to sync_io::stat's declare_stats() (found via ADL on sync_io::stat::Channel_stats).
147 * The sync_io stats keep `name_prefix` unchanged (no "core." prefix added): they are meaningful in their own
148 * right, and the context/separation should be quite clear visually. */
149 sync_io::stat::declare_stats(name_prefix,
150 src_stats ? &src_stats->m_core : nullptr,
151 target_stats ? &target_stats->m_core : nullptr,
152 visitor);
153
154 /* m_sync_req: also delegate (it is reset standalone via Channel::stats_reset()).
155 * Add "sync_req." sub-prefix; as of this writing without the "sync_req." context these stats look rather
156 * ambiguous at a glance; for example `count=[4]`: that is count of sync_request()s. Hopefully you feel
157 * the distinction between that and hypothetically adding `core.` (or whatever) above. */
158 declare_stats(flow::util::stat::Stat_name{name_prefix, "sync_req."},
159 src_stats ? &src_stats->m_sync_req : nullptr,
160 target_stats ? &target_stats->m_sync_req : nullptr,
161 visitor);
162} // declare_stats(Channel_stats)

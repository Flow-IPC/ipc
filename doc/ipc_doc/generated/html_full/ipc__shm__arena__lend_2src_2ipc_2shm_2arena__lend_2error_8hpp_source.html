106/**
107 * Ummm -- it specializes this `struct` to -- look -- the end result is boost.system uses this as
108 * authorization to make `enum` `Code` convertible to `Error_code`. The non-specialized
109 * version of this sets `value` to `false`, so that random arbitrary `enum`s can't just be used as
110 * `Error_code`s. Note that this is the official way to accomplish that, as (confusingly but
111 * formally) documented in boost.system docs.
112 */
113template<>
114struct is_error_code_enum<::ipc::shm::arena_lend::error::Code>
115{
116 /// Means `Code` `enum` values can be used for `Error_code`.
117 static const bool value = true;
118 // See note in similar place near transport::error.
119};

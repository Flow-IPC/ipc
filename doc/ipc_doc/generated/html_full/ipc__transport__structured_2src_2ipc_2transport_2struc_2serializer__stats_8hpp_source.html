41/**
42 * Simple bucket configuration for one `flow::util::stat::Histogram_counter` inside Serializer_stats::Snd et al;
43 * aggregate-initializable. build() returns a fresh `Histogram_counter`. Two scales are expressible:
44 * - *Linear* (#m_ratio zero): `{n_buckets, bucket0_sz, bucket_sz}`, straight into the like-named
45 * `Histogram_counter` ctor args (with `bucket0_val0` always 0): bucket 0 is `[0, m_bkt0)`; each
46 * subsequent bucket is #m_bkt wide.
47 * - *Geometric* (#m_ratio at least 2): `{n_buckets, floor, 0, ratio}`: bucket 0 is `[0, floor)` -- the
48 * catch-all below the ladder -- and each subsequent bucket spans `ratio` times its predecessor:
49 * bucket 1 = `[floor, floor * ratio)`, etc. (As always values beyond the last bucket land in it
50 * regardless.) This scale suits open-ended value regimes spanning several orders of magnitude, where
51 * any practical linear scale would lump everything past its reach into the last bucket.
52 */
53struct Histo_cfg
54{
55 // Data.
56
57 /// See Histogram_counter ctor (either scale; total bucket count including the bucket-0 catch-all).
58 size_t m_n_bkts;
59 /// Linear scale: see Histogram_counter ctor; geometric scale: the ladder floor (bucket 0 = `[0, m_bkt0)`).
60 size_t m_bkt0;
61 /// Linear scale: see Histogram_counter ctor; geometric scale: meaningless; must be zero.
62 size_t m_bkt;
63 /// Zero => linear scale; else (then it must be >= 2) => geometric scale with this bucket-to-bucket size ratio.
64 size_t m_ratio = 0;
65
66 // Methods.
67
68 /**
69 * Builds a fresh Histogram_counter using these knobs; see `struct` doc header for their semantics per scale.
70 * @return See above.
71 */
72 flow::util::stat::Histogram_counter build() const;
73}; // struct Histo_cfg

75/**
76 * Cumulative stats describing capnp-driven memory-allocation activities on behalf of a struc::Struct_builder +
77 * struc::Struct_reader impl pair (usually <=> Msg_out and Msg_in respectively), such that they both belong
78 * to a conceptually similar use-case for needing memory-for-capnp. That is admittedly very abstract-sounding
79 * and arguably is best explained by giving some background and a practical example.
80 *
81 * @see Msg_out and Msg_in doc headers.
82 * @see Struct_builder concept doc header (also Struct_reader as desired).
83 * @see Doc headers for impls thereof: Heap_fixed_builder, Heap_reader; shm::Builder, shm::Reader.
84 *
85 * It is possible to write custom impls of Struct_builder and Struct_reader, and it is possible (albeit not
86 * the envisioned primary use case) to use any of them independently of Msg_out and Msg_in (and struc::Channel)
87 * respectively. Even then, the existing impls, used with our Msg_out and Msg_in, are the canonical/reference
88 * situation. So we assume it for this doc header.
89 *
90 * In short: there is, first, the sender side. One creates Msg_out, which creates Struct_builder (a specific
91 * impl) on its own behalf. One capnp-mutates it via capnp-generated setters and other mutators. When
92 * desired (which can be never, or once, or many times) one sends it via, e.g., Channel::send().
93 *
94 * In short: there is, also, the receiver side. On receipt of some representation of a `Msg_out`-at-send-time,
95 * struc::Channel (et al) creates Msg_in which creates Struct_reader (a specific impl) on its own behalf.
96 * It reads aspects of the serialization received via IPC and makes-available a (read-only) capnp-generated
97 * accessor API for reading the contents of the Msg_in by the user.
98 *
99 * Now consider the two canonical available `Struct_builder`s and counterpart `Struct_reader`s. Note that
100 * Msg_out and Msg_in, respectively, are template-parameterized on the builder and reader impl.
101 *
102 * ### Pure-heap builder/reader pair: Heap_fixed_builder, Heap_reader ###
103 * There is just the outer serialization; it is directly mutated by the user (send-side) and read by the user
104 * (receive-side), with a full copy occurring when the bytes travel into an IPC-transport (at send time) and
105 * then out of it on the opposing side (different process probably).
106 *
107 * Msg_out / Heap_fixed_builder are used for capnp-mutation (by user) causing ongoing allocation (by builder
108 * internals). This is tracked by Snd: #m_snd.
109 *
110 * Msg_in / Heap_reader are used for capnp-access (by user), following one-time allocation and fill-out of
111 * required segments as received via IPC. This is tracked by Rcv: #m_rcv.
112 *
113 * So a `*this`, in this context, tracks counts of events and gauges of currently-outstanding resources
114 * (e.g., messages or allocated or serialization-used bytes therein), split up into those on behalf of the
115 * *sender-side* (so Heap_fixed_builder) aspects, and the *receiver-side* (so Heap_reader) ones.
116 *
117 * ### SHM-backed builder/reader pair: shm::Builder, shm::Reader ###
118 * Please see doc header for Rcv: section Background. In short:
119 * - shm::Builder internally keeps a Heap_fixed_builder, and shm::Reader a Heap_reader, because the
120 * outer-serialization, containing exclusively a 1-segment capnp-message holding a SHM-handle to the
121 * core/inner-serialization, is still otherwise-identically used: allocate, mutate, send, receive,
122 * allocate, access.
123 * - That is one `*this` with #m_snd and #m_rcv.
124 * - The true, user-mutated (sender-side) and user-accessed (receiver-side) capnp data (the
125 * core/inner-serialization) is in SHM.
126 * - That is a 2nd, separate `*this` with #m_snd -- and an unused (zeroed-out) #m_rcv. The latter
127 * is because no in-SHM allocation is required on receipt; that's the point of SHM.
128 *
129 * The key thing to grasp is that while shm::Builder *wraps* both a Heap_fixed_builder again *and* code
130 * that allocates in SHM, the stats corresponding to each of the two types of in-memory stuff -- for
131 * pragmatic reasons -- are tracked separately, in separate `Serializer_stats`es. Quantities in one are
132 * not necessarily a subset of the respective ones in the other.
133 *
134 * ### Key aliases ###
135 * Three Serializer_stats variants exist in Flow-IPC as of this writing, each with a ready-to-use
136 * default-constructible alias:
137 *
138 * - struc::stat::Heap_serializer_stats: pure-heap user messages (alias defined in struc_fwd.hpp).
139 * - struc::shm::stat::Outer_serializer_stats: heap-backed outer-serialization for a SHM-backed user message
140 * (low byte volume; each message has 1 segment of a constant, tiny size).
141 * - struc::shm::stat::Core_serializer_stats: SHM-backed core/inner-serialization (actual user payload).
142 * - To repeat: #m_rcv is unused/all-zeroes.
143 *
144 * That said, these stat-set `struct`s can be used in other, custom contexts as desired.
145 *
146 * Concurrency
147 * -----------
148 * In the mainstream use-cases above, stat-updating can occur from multiple threads and concurrently to
149 * potential consumption by the user (you). General heap-or-SHM allocation can *potentially* be too
150 * frequent to keep un-sharded stats; but in the Msg_out / Msg_in / Struct_builder / Struct_reader context
151 * updates shall not be so frequent as to cause significant contention. Therefore all non-histogram stat
152 * members are `atomic`. (`Histogram_counter` is internally `atomic`, in that each bucket's count-increment
153 * and loads are atomic.) Mainstream stat-updating is expected to be done on globally-maintained
154 * Serializer_stats objects.
155 *
156 * @see `flow::util::stat` namespace doc header for an in-depth discussion of multi-thread stat-keeping
157 * and perf.
158 *
159 * Any mechanism for maintaining such globals is fine; but we recommend (and Flow-IPC actively uses)
160 * `flow::util::stat::Stat_set_list` and its singleton-providing wrapper `flow::util::stat::Global_stats`.
161 *
162 * Why concrete-type Serializer_stats + template Serializer_stats_p?
163 * -----------------------------------------------------------------
164 * Almost all relevant code -- whether stat-producer (the builders and readers) or stat-consumer (you) --
165 * only accesses each Serializer_stats via its #m_snd and #m_rcv parts after creation. E.g.,
166 * Heap_fixed_builder is given a pointer to an #m_snd and increments Snd::m_msgs at the proper points.
167 *
168 * Each Serializer_stats has to be stored somewhere, though, and first initialized. Most members are
169 * trivially init'd at declaration. However, certain `Histogram_counter`s have a number/width of buckets
170 * that depends on the context (e.g., shm::stat::Core_serializer_stats messages can be huge, while
171 * shm::stat::Outer_serializer_stats ones are tiny). So Serializer_stats's ctor takes Histo_cfg knobs (as do
172 * #m_snd's and #m_rcv's ctors).
173 *
174 * However, in the mainstream, the stats here are expected to be kept as globals via `Global_stats`, which
175 * requires its `Stat_set` template-parameter type to be default-constructible. As noted, our
176 * Serializer_stats is not (otherwise some histograms would be useless in some cases). So
177 * Serializer_stats_p<Cfg> is a thin templated subclass with no extra data that default-constructs by
178 * passing `Cfg::S_HISTO_*` Histo_cfg constants into the base ctor. Pointers/refs cast safely between the
179 * two.
180 *
181 * The key tactical point is:
182 * - ~All stat-consuming/-producing code deals with Serializer_stats (no template) ptrs/refs.
183 * - When specifying the `Global_stats::Stat_set` template-parameter, use Serializer_stats_p (template),
184 * with the desired knobs for the particular type-of-builder-and-reader. Then, on calling
185 * `flow::util::stat::Stat_set_list::stats_[mutable_][_at|_default]()`, safely up-cast the returned ref
186 * to `[const] Serializer_stats&`.
187 */
188struct Serializer_stats
189{

192 /**
193 * Send-side (during-serialization) stats, captured at hooks within the Struct_builder
194 * impl: typically each `capnp::MessageBuilder::allocateSegment()` call and the builder's destructor.
195 *
196 * Context: Typically `Struct_builder`s are used as core components of `Msg_out`s, which are in turn
197 * sent via struc::Channel and struc::sync_io::Channel; though Struct_builder impls are available
198 * for direct use also. See Serializer_stats doc header for more such context.
199 */
200 struct Snd
201 {
202 // Data.
203
204 /**
205 * Total `Struct_builder` (usually <=> Msg_out) instances ever that triggered 1+ allocation through the
206 * builder. Distinct from `Channel`-layer message counts: that one counts *send invocations*,
207 * this one counts *builder instances*. A single Msg_out can be sent more than once.
208 */
209 std::atomic<uint64_t> m_msgs = 0;
210
211 /**
212 * Gauge: How many of items counted by #m_msgs are currently-alive: increment at first-alloc, decrement
213 * at `Struct_builder` (usually <=> Msg_out) dtor or move-from.
214 */
215 std::atomic<size_t> m_msgs_outstanding = 0;
216
217 /// High water-mark of #m_msgs_outstanding observed so far.
218 std::atomic<size_t> m_msgs_outstanding_hi_wmark = 0;
219
220 /**
221 * Lifetime sum of allocated bytes across all `capnp::MessageBuilder::allocateSegment()` calls (over all items
222 * counted in #m_msgs). Excludes frame-prefix bytes (covered separately by #m_frame_lifetime_sz). Includes
223 * past-end slack within each segment (i.e., counts the segment's allocated capacity, not just the bytes
224 * capnp ends up using).
225 *
226 * @note The aforementioned "past-end slack" might be quite high for a given segment at first; then as one
227 * mutates the capnp structure in standard fashion, the slack decreases, until either the whole thing
228 * is destroyed, or there's no space for some new datum, and another segment is created to hold that
229 * one. We modify #m_alloc_lifetime_sz in `allocateSegment()`; no notion of "past-end slack" is
230 * usefully measurable at that point. Grabbing it per-segment in builder dtor/move-from is possible but is
231 * not what we do here.
232 *
233 * Total physical heap allocation = #m_alloc_lifetime_sz plus #m_frame_lifetime_sz.
234 */
235 std::atomic<uint64_t> m_alloc_lifetime_sz = 0;
236
237 /// Gauge: Of bytes tracked by #m_alloc_lifetime_sz, how many are currently allocated as opposed to now-freed?
238 std::atomic<size_t> m_alloc_outstanding_sz = 0;
239
240 /// High water-mark of #m_alloc_outstanding_sz observed so far.
241 std::atomic<size_t> m_alloc_outstanding_sz_hi_wmark = 0;
242
243 /**
244 * Per-`Struct_builder` (usually <=> Msg_out) histogram of total allocated bytes through the builder
245 * (see #m_alloc_lifetime_sz regarding inclusion semantics) -- sampled at end-of-life of the builder
246 * (`Struct_builder` dtor or move-from).
247 *
248 * Builders that have not yet been destroyed are absent from this histogram (use #m_msgs_outstanding /
249 * #m_msgs_outstanding_hi_wmark to detect those).
250 */
251 flow::util::stat::Histogram_counter m_histo_msg_alloc_sz;
252
253 /**
254 * Per-`Struct_builder` (usually <=> Msg_out) histogram of bytes actually used by capnp (sum of
255 * `getSegmentsForOutput()` segment `.size()`s) -- sampled at end-of-life of the builder
256 * (`Struct_builder` dtor or move-from). Slack between #m_histo_msg_alloc_sz and this histogram = capnp's
257 * internal past-end waste (space where it could not fit mutated-in data and chose to go to a new segment instead).
258 */
259 flow::util::stat::Histogram_counter m_histo_msg_used_sz;
260
261 /**
262 * Per-`Struct_builder` (usually <=> Msg_out) histogram of segment count -- sampled at end-of-life of the builder
263 * (`Struct_builder` dtor or move-from). Buckets: [1], [2], [3+].
264 */
265 flow::util::stat::Histogram_counter m_histo_segs_per_msg{3, 1, 1, 0};
266
267 /**
268 * Lifetime count of `capnp::MessageBuilder::allocateSegment()` calls in which the leaf-driven minimum size
269 * requested by capnp exceeded the builder's preferred next-segment size at that point. An operator signal:
270 * "users have leaves bigger than your starting or exponential-growth-driven segment size -- consider tuning
271 * the initial-segment size up and/or looking into the design of your schema." It is not necessarily a
272 * terrible thing; depends; but knowing about big leaves is good. In capnp a big leaf as of this writing
273 * can only be a `List`, `Data`, or `Text` field.
274 */
275 std::atomic<uint64_t> m_big_leaf_alloc_count = 0;
276
277 /**
278 * Per-event histogram of leaf-driven minimum size as passed to `capnp::MessageBuilder::allocateSegment()`
279 * (but in bytes!) for each event counted in #m_big_leaf_alloc_count. Answers "how big are those big leaves?"
280 */
281 flow::util::stat::Histogram_counter m_histo_big_leaf_sz;
282
283 /**
284 * `Struct_builder` (usually <=> Msg_out) count, where each builder experienced at least 1 big-leaf event
285 * (counted in #m_big_leaf_alloc_count). Distinguishes "one outlier message with N big leaves" from
286 * "X% of messages have at least 1 big leaf."
287 */
288 std::atomic<uint64_t> m_msgs_with_big_leaf = 0;
289
290 /**
291 * `Struct_builder` (usually <=> Msg_out) count of builders observing the *capacity-lock* event:
292 * in `capnp::MessageBuilder::allocateSegment()`, when computing the ~exponentially-growing next-segment
293 * size (used only if there *is* another `allocateSegment()`), the value was clamped to the configured max.
294 * The event is counted regardless of whether that next `allocateSegment()` does occur or not.
295 *
296 * This event
297 * - can happen once, or not at all, per builder;
298 * - as of this writing is not observed in the shm::Builder core (in-SHM) allocating algorithm, as the
299 * cap pragmatically exists to avoid exceeding an IPC-transport's single-message size -- a concern that
300 * is irrelevant to in-SHM (or equivalent zero-copy) allocations.
301 */
302 std::atomic<uint64_t> m_seg_grow_cap_lock_count = 0;
303
304 /**
305 * Lifetime sum of frame-prefix bytes across all `capnp::MessageBuilder::allocateSegment()` calls.
306 *
307 * The frame prefix is a (usually small, in fact for most segments zero in practice) configurably-sized
308 * area reserved in front of the actual capnp-segment space, when the buffer for the segment is allocated.
309 * So the frame prefix (if any) can hold some data maintained by someone else; capnp segment begins after
310 * that.
311 */
312 std::atomic<uint64_t> m_frame_lifetime_sz = 0;
313
314 // Constructors.
315
316 /**
317 * Constructor; supplies bucket-structure knobs for the `Histogram_counter`s in `*this` whose useful shape
318 * varies by use-case. E.g., in-SHM messages can plausibly be much larger than IPC-transport-constrained
319 * heap-based ones; so a Heap_fixed_builder versus shm::Builder might use different knobs here.
320 *
321 * @param msg_alloc_sz_cfg
322 * For #m_histo_msg_alloc_sz.
323 * @param msg_used_sz_cfg
324 * For #m_histo_msg_used_sz.
325 * @param big_leaf_sz_cfg
326 * For #m_histo_big_leaf_sz.
327 */
328 Snd(Histo_cfg msg_alloc_sz_cfg, Histo_cfg msg_used_sz_cfg, Histo_cfg big_leaf_sz_cfg);
329 }; // struct Snd

331 /**
332 * Receive-side (during-deserialization) stats, captured at hooks within the Struct_reader impl.
333 *
334 * Context: Typically `Struct_reader`s are used as core components of `Msg_in`s, which are in turn
335 * received (and then given to the user via on-receive handlers) via struc::Channel and
336 * struc::sync_io::Channel; though Struct_reader impls are available for direct use also. See
337 * Serializer_stats doc header for more such context.
338 *
339 * ### Background / This is all zeroes for in-SHM-related stats ###
340 * This is a bit subtle, if one is not fully experienced in how builders/readers work (Struct_builder concept
341 * doc header is good background reading), so read carefully:
342 *
343 * Consider Heap_fixed_builder, as used in the most straightforward, purely-heap-backed setup. Let's assume
344 * we're in the most-mainstream context (with Msg_out). When the Msg_out is populated/modified/destroyed,
345 * that's Snd. If this message is sent -- which can be done 0, 1, 2... times over any
346 * suitable `Channel` -- then on the receiving side, the resulting capnp-serialization is copied out of the
347 * IPC transport. For this, buffer(s) is/are allocated (again, in possibly different process(es)). That's
348 * Rcv.
349 *
350 * Now consider shm::Builder, the recommended zero-copy-friendly Struct_builder. There are now two
351 * serializations: the core/inner one; and the outer one, where the SHM-handles-to-the-inner-serialization-
352 * segments-in-SHM are encoded. So shm::Builder uses *two* types of allocation on the *builder* (sending)
353 * side:
354 * - the Heap_fixed_builder for the outer serialization (in practice small and simple: per message, always
355 * 1 segment of a constant small size enough for a SHM-handle only);
356 * - the shm::Builder allocating *actual* user data in actual SHared Memory (SHM).
357 *
358 * That's tracked, again, by Snd, but two of them: one for the latter (more interesting) and
359 * one for the former (probably less interesting).
360 *
361 * Now the punchline: when a message is received, all of the above regarding Rcv and
362 * shm::Reader still holds... except that the whole point of the *core/inner* (in-SHM) storage is that it
363 * need be done only once; there's no copying, so no in-SHM allocating occurs on receipt. Therefore:
364 * - the Heap_reader still keeps stats in a Rcv; *but*:
365 * - there is nothing that would update the Rcv for the directly-in-SHM items. So they
366 * stay zero in the containing Serializer_stats.
367 *
368 * We could have structured things a bit differently (less symmetrically) to avoid this; but the resulting
369 * reduction in boilerplate is arguably worth the trade-off.
370 */
371 struct Rcv
372 {
373 // Data.
374
375 /**
376 * Gauge: Currently-alive `Struct_reader` (usually <=> Msg_in) instances.
377 *
378 * @internal
379 * Incremented at deserialization-start time, decremented at reader dtor.
380 * @endinternal
381 */
382 std::atomic<size_t> m_msgs_outstanding = 0;
383
384 /// High water-mark of #m_msgs_outstanding observed so far.
385 std::atomic<size_t> m_msgs_outstanding_hi_wmark = 0;
386
387 /**
388 * Lifetime sum of segment-blob capacity across all `Struct_reader` (usually <=> Msg_in) instances ever.
389 * Sampled at a given reader's deserialization-start time, same point as when incrementing
390 * #m_msgs_outstanding.
391 *
392 * Certain complicating niggles aside, it works basically like this inside at least sync_io::Channel:
393 * -# Need to receive a segment; prepare definitely-big-enough buffer: N bytes.
394 * -# Receive it over a transport::Blob_receiver (e.g., transport::Native_socket_stream): message was
395 * M<=N bytes.
396 * -# OK, this is (a) segment in the capnp-serialization we are receiving: M useful bytes; (N minus M)
397 * unused bytes.
398 *
399 * Here we are tracking the N, not the M.
400 */
401 std::atomic<uint64_t> m_alloc_lifetime_sz = 0;
402
403 /// Gauge: Of bytes tracked by #m_alloc_lifetime_sz, how many are currently allocated as opposed to now-freed?
404 std::atomic<size_t> m_alloc_outstanding_sz = 0;
405
406 /// High water-mark of #m_alloc_outstanding_sz observed so far.
407 std::atomic<size_t> m_alloc_outstanding_sz_hi_wmark = 0;
408
409 /**
410 * Gauge: Subset of #m_alloc_outstanding_sz that contains actual capnp-serialization payload.
411 *
412 * @see Doc header for #m_alloc_lifetime_sz: Here we track M. See #m_alloc_outstanding_sz which
413 * tracks N. The difference between the two is the total unused (harshly: wasted) space at this time.
414 */
415 std::atomic<size_t> m_used_outstanding_sz = 0;
416
417 /// High water-mark of #m_used_outstanding_sz observed so far.
418 std::atomic<size_t> m_used_outstanding_sz_hi_wmark = 0;
419
420 /// Histogram tracking per-`Struct_reader` (usually <=> Msg_in) contribution to #m_alloc_lifetime_sz.
421 flow::util::stat::Histogram_counter m_histo_msg_alloc_sz{9, 64 * 1024, 64 * 1024, 0};
422
423 /// Subset of bytes tracked in #m_histo_msg_alloc_sz that contains actual capnp-serialization payload.
424 flow::util::stat::Histogram_counter m_histo_msg_used_sz;
425
426 // Constructors.
427
428 /**
429 * Constructor; supplies bucket-structure knobs for the variable-shape `Histogram_counter`(s) in `*this`.
430 * See Snd::Snd() ctor doc header; same explanation applies here.
431 *
432 * @param msg_used_sz_cfg
433 * For #m_histo_msg_used_sz.
434 */
435 explicit Rcv(Histo_cfg msg_used_sz_cfg);
436 }; // struct Rcv

464/**
465 * Templated no-data, non-polymorphic subclass of Serializer_stats whose precipitating use-case is to satisfy
466 * `flow::util::stat::Global_stats<Stat_set>`'s default-constructibility requirement for `Stat_set`.
467 *
468 * @see Serializer_stats doc header, section "Why concrete-type...". It explains that we exist to take #Cfg
469 * at compile-time, so the concrete type's ctor can require no args. It further explains that
470 * all stat-keeping and stat-updating code would be working on a `Serializer_stats*` (or `&`), obtained
471 * via safe up-cast from a `Serializer_stats_p<...knobs...>`.
472 *
473 * @tparam Cfg_t

177TEMPLATE_CLSC_CLI_SESSION_IMPL
178template<typename Task_err>
179bool CLASS_CLSC_CLI_SESSION_IMPL::async_connect
180 (const typename Base::Base::Base::Mdt_builder_ptr& mdt,
181 typename Base::Base::Base::Channels* init_channels_by_cli_req_pre_sized,
182 typename Base::Base::Base::Mdt_reader_ptr* mdt_from_srv_or_null,
183 typename Base::Base::Base::Channels* init_channels_by_srv_req,
184 Task_err&& on_done_func)
185{
186 using boost::make_shared;
187 using boost::shared_ptr;
188 using Channels = typename Base::Base::Base::Channels;
189 using Mdt_reader_ptr = typename Base::Base::Base::Mdt_reader_ptr;
190
191 /* Very simple really: server is in charge of setting up the SHM arenas; we just open them.
192 * Unlike on the server side, no one but the us-calling user needs to be gated on our successfully opening
193 * them; so we can simply perform the entire vanilla Base::Base::async_connect(), and if that went fine then
194 * open the arenas.
195 *
196 * The only wrinkle is: what if (unlikely though it is) the opening fails? *(Base*)this is already in PEER
197 * state by then. Well, cancel_peer_state_to_null() exists for that purpose. A bit cheesy? One could say that...
198 * but it's not that bad.
199 *
200 * Okay, another wrinkle; don't set the out-args, until we are really successful; so put stuff
201 * into some intermediate targets first. */
202
203 auto temp_init_channels_by_cli_req_pre_sized
204 = init_channels_by_cli_req_pre_sized ? make_shared<Channels>(init_channels_by_cli_req_pre_sized->size())
205 : shared_ptr<Channels>();
206 auto temp_mdt_from_srv_or_null
207 = mdt_from_srv_or_null ? make_shared<Mdt_reader_ptr>() : shared_ptr<Mdt_reader_ptr>();
208 auto temp_init_channels_by_srv_req
209 = init_channels_by_srv_req ? make_shared<Channels>() : shared_ptr<Channels>();
210
211 return Base::Base::async_connect(mdt,
212 temp_init_channels_by_cli_req_pre_sized.get(),
213 temp_mdt_from_srv_or_null.get(),
214 temp_init_channels_by_srv_req.get(),
215 [this, on_done_func = std::move(on_done_func),
216 init_channels_by_cli_req_pre_sized, mdt_from_srv_or_null, init_channels_by_srv_req,
217 temp_init_channels_by_cli_req_pre_sized, // @todo Avoid s_p<> copies here.
218 temp_mdt_from_srv_or_null,
219 temp_init_channels_by_srv_req]
220 (const Error_code& async_err_code)
221 {
222 if (async_err_code)
223 {
224 // No questions asked. GTFO.
225 on_done_func(async_err_code);
226 return;
227 }
228 /* else: Cool; just do what shm::classic::Server_session_impl did except much simpler:
229 * - Merely open, no creating.
230 * - Lifetime of the Arena (pool handle) equals `*this`; so no need to interface with any daddy
231 * server object; just do it. */
232
233 auto err_code = async_err_code; // We may need to modify it anyway below, so make a copy.
234 const auto srv_app_name = Shared_name::ct(Base::Base::Base::m_srv_app_ref.m_name);
235 const auto cli_app_name = Shared_name::ct(Base::Base::Base::cli_app_ptr()->m_name);
236 const auto& srv_namespace = Base::Base::Base::srv_namespace();
237
238 Arena_ptr session_shm;
239 Arena_ptr app_shm;
240 auto shm_pool_name = build_conventional_shared_name(Shared_name::S_RESOURCE_TYPE_ID_SHM,
241 srv_app_name, srv_namespace, cli_app_name,
242 Base::Base::Base::cli_namespace())
243 / SHM_SUBTYPE_PREFIX
244 / Shared_name::S_1ST_OR_ONLY;
245 session_shm = make_shared<typename Base::Arena>
246 (get_logger(), shm_pool_name, util::OPEN_ONLY, false, &err_code);
247
248 if (!err_code)
249 {
250 shm_pool_name = build_conventional_shared_name(Shared_name::S_RESOURCE_TYPE_ID_SHM,
251 srv_app_name, srv_namespace, cli_app_name)
252 / SHM_SUBTYPE_PREFIX
253 / Shared_name::S_1ST_OR_ONLY;
254
255 app_shm = make_shared<typename Base::Arena>
256 (get_logger(), shm_pool_name, util::OPEN_ONLY, false, &err_code);
257 } // if (!err_code) (but may have become truthy inside)
258
259 if (err_code)
260 {
261 /* Get back to NULL state all around. Note: do *not* ever clean underlying shared resources (server's job!).
262 * Careful: a failed open-only Pool_arena ctor still yields a (non-null) object in a failed state; so the
263 * locals session_shm and/or app_shm may be non-null here. They go away on their own; and we never
264 * init_shm_arenas() on failure, so the next connect attempt starts from scratch. */
265 Base::Base::cancel_peer_state_to_null();
266 }
267 else
268 {
269 // Well... great! Stay in PEER state.
270 Base::init_shm_arenas(std::move(session_shm), std::move(app_shm));
271
272 // Finalize out-args.
273 if (init_channels_by_cli_req_pre_sized)
274 {
275 *init_channels_by_cli_req_pre_sized = std::move(*temp_init_channels_by_cli_req_pre_sized);
276 }
277 if (mdt_from_srv_or_null)
278 {
279 *mdt_from_srv_or_null = std::move(*temp_mdt_from_srv_or_null);
280 }
281 if (init_channels_by_srv_req)
282 {
283 *init_channels_by_srv_req = std::move(*temp_init_channels_by_srv_req);
284 }
285 } // else if (!err_code)
286
287 on_done_func(err_code); // err_code could be truthy.
288 }); // return Base::Base::async_connect()
289} // Client_session_impl::async_connect()
